### 9.5 `src/main.ts`（主进程纯函数）

```typescript
import type { BaseCardView } from '@chips/sdk';

const DEFAULT_FONT_SIZE = 16;
const DEFAULT_LINE_HEIGHT = 1.6;
const MIN_FONT_SIZE = 12;
const MAX_FONT_SIZE = 48;

interface IRichTextConfig {
  fontSize: number;
  lineHeight: number;
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

export function validateConfig(config: unknown): boolean {
  if (config === undefined || config === null) {
    return true;
  }
  if (!isRecord(config)) {
    return false;
  }
  const fontSize = config.fontSize;
  if (
    fontSize !== undefined &&
    (typeof fontSize !== 'number' ||
      !Number.isInteger(fontSize) ||
      fontSize < MIN_FONT_SIZE ||
      fontSize > MAX_FONT_SIZE)
  ) {
    return false;
  }
  const lineHeight = config.lineHeight;
  if (lineHeight !== undefined && (typeof lineHeight !== 'number' || lineHeight < 1)) {
    return false;
  }
  return true;
}

export function createInitialContent(): Record<string, unknown> {
  return {
    card_type: 'base.richtext',
    content_format: 'markdown',
    content_source: 'inline',
    content_text: '',
    config: {
      fontSize: DEFAULT_FONT_SIZE,
      lineHeight: DEFAULT_LINE_HEIGHT,
    },
  };
}

export function collectResourcePaths(basecard: BaseCardView): string[] {
  void basecard;
  // 富文本内容不引用包内资源；如类型支持图片，应在此返回去重、稳定排序的包内相对路径。
  return [];
}
```
