### 8.1 主逻辑注册与处理（main 目标）

```typescript
import type { IAppContext, IAppPlugin } from '@chips/sdk';
import { ChipsError } from '@chips/sdk';

const COMMAND_ID = 'iconmaker:generate';

        title: '生成图标',
        description: '从当前图片生成 PNG、ICO 与 ICNS 图标包',
        icon: 'assets/icons/generate.svg',
        shortcut: 'CommandOrControl+Alt+G',
        scope: { kind: 'app' },
        when: { enabled: { key: 'workspaceOpen', truthy: true } },
        paramsSchema: {
          type: 'object',
          additionalProperties: false,
          required: ['source'],
          properties: {
            source: { type: 'string', minLength: 1 },
            overwrite: { type: 'boolean', default: false },
          },
        },
        keywords: ['图标', '生成', 'icns'],
        group: '导出',
      },
    });
  }

  public async handleCommand(command: string, args: unknown): Promise<void> {
    if (command !== COMMAND_ID) {
      throw new ChipsError({
        code: 'COMMAND_NOT_FOUND',

      });
    }
```

### 8.2 UI 注册、本地处理器与命令面板消费（ui 目标）

```tsx
import React from 'react';
import { chips } from '@chips/sdk';
import { CommandPalette, CommandProvider, useCommandRuntime } from '@chips/ui';

const COMMAND_ID = 'player:toggle-play';
const HANDLER_ID = 'toggle-play';

const PlayerCommands: React.FC = () => {
  const runtime = useCommandRuntime();

  React.useEffect(() => {
    void chips.request('command:register', {
      definition: {
        id: COMMAND_ID,
        title: '播放/暂停',
        icon: 'assets/icons/play.svg',
        shortcut: 'Space',
        scope: { kind: 'app' },
        when: {
          visible: true,
          enabled: { key: 'windowKind', equals: 'window' },
        },
        handlerId: HANDLER_ID,
        group: '播放',
      },
    });

    const attachment = runtime.attach([
      {
        commandId: COMMAND_ID,
        handlerId: HANDLER_ID,
        handler: () => {
          // 切换播放状态，更新本地 UI
        },
      },
    ]);

    return () => {
      attachment.dispose();
      void chips.request('command:unregister', { commandId: COMMAND_ID });
    };
  }, [runtime]);

  return null;
};

export const AppView: React.FC = () => (
  <CommandProvider>
    <PlayerCommands />
    <CommandPalette />
  </CommandProvider>
);
```

### 8.3 菜单与快捷键引用同一记录

```typescript
import type { IAppContext } from '@chips/sdk';

async function buildMenus(ctx: IAppContext): Promise<void> {
  // 命令记录已在注册表中（如 editor:save）
  await ctx.bridge.request('menu:set', {
    target: 'app',
    items: [
      {
        id: 'file',
        label: '文件',
        submenu: [
          { id: 'save', label: '保存', command: 'editor:save' },
        ],
      },
    ],
  });
}
```

`editor:save` 的定义由编辑器应用注册，其中声明 `shortcut: 'CommandOrControl+S'`；菜单项触发时平台以 `source: 'menu'` 调用 `command:invoke`，窗口内按 Cmd/Ctrl+S 时以 `source: 'shortcut'` 调用同一记录，两条入口共用同一校验、同一审计。

### 8.4 CLI 共享同一记录

CLI 入口声明（格式以 [CLI 完整参考](../06-开发与运维/07-CLI完整参考.md) 为准）的 `commandId` 指向 `iconmaker:generate`；CLI 执行时平台以 `source: 'cli'` 调用：

```typescript
await ctx.bridge.request('command:invoke', {
  commandId: 'iconmaker:generate',
  payload: { source: '/tmp/icon.png', overwrite: true },
  source: 'cli',
  context: { appId: 'chips.official.iconmaker' },
});
```

若 `chips.official.iconmaker` 尚未运行，CLI 先按正式启动链路以 `source: command` 启动应用实例并等待 `iconmaker:generate` 注册，再执行上述调用；等待、进度与产物输出使用 CLI 任务通道。
