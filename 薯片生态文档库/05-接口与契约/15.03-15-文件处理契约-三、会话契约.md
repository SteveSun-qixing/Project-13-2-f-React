# 文件处理契约

> 版本：v1.0.0
> 日期：2026-09-19
> 状态：正式版
> 适用范围：通用文件处理（`.card` / `.box` 文档之外的普通文件）的会话与转换编排契约：`file-processing` 服务、打开/预览/转换/另存/比较操作、进度与取消、受控提交与原子替换、大文件流式、错误码与边界；不定义具体格式算法、支持矩阵与原子能力实现

---

## 三、会话契约

### 3.1 类型总表

```typescript
import type { Dispose } from '@chips/sdk';

/** 文件处理会话操作。 */
type FileProcessOperation = 'preview' | 'convert' | 'save-as' | 'compare';

/** 会话状态。 */
type FileProcessingSessionState = 'opened' | 'busy' | 'closed';

/** 编排阶段。 */
type FileProcessStage =
  | 'prepare'      // 探测来源、选择提供方、准备暂存区
  | 'transform'    // 内容处理（可多次原子步骤）
  | 'stage'        // 暂存产物落盘与校验
  | 'publish'      // 受控提交：原子发布到目标
  | 'cleanup'      // 清理暂存产物
  | 'completed';   // 终态

/** 进度载荷。 */
interface IFileProcessProgress {
  operation: FileProcessOperation;
  stage: FileProcessStage;
  /** 同一操作内 0-100、单调不减。 */
  percent: number;
  /** 简体中文、面向用户。 */
  message?: string;
}

/** 来源信息。 */
interface IFileProcessingSourceInfo {
  /** 受管引用（规范形式）。 */
  uri: string;
  /** 探测得到的 MIME。 */
  mimeType: string;
  /** 规范化格式标识（如 card、docx、pdf、mp4）；探测失败为 unknown。 */
  format: string;
  size: number;
  /** 来源版本戳；变化即视为外部修改。 */
  version: string;
}

/** 打开会话输入。 */
interface IFileProcessingOpenInput {
  /** 受管引用：chips-resource:// 或按 06 §2.6 归一的 file:// / http(s):// / webdav://。 */
  source: string;
  signal?: AbortSignal;
}

/** 产物目标。 */
interface IFileProcessTarget {
  /** 受管引用；目标作用域必须为调用方持有写权限的可写作用域。 */
  uri: string;
  /** 目标已存在时的覆盖策略；默认 false。 */
  overwrite?: boolean;
}

/** 格式选项。公共字段之外的提供方专有选项放 formatOptions。 */
interface IFileProcessFormatOptions {
  quality?: number;                          // 1-100，有损编码
  width?: number;
  height?: number;
  fit?: 'contain' | 'cover';
  pageSize?: 'A4' | 'A3' | 'Letter' | 'Legal';
  landscape?: boolean;
  background?: 'transparent' | 'white';
  page?: number;                             // 1 起
  timeSeconds?: number;                      // ≥ 0
  locale?: string;
  formatOptions?: Readonly<Record<string, string | number | boolean>>;
}

/** 预览输入。 */
interface IFilePreviewInput {
  width?: number;
  height?: number;
  fit?: 'contain' | 'cover';
  format?: 'png' | 'jpeg' | 'webp';          // 默认 png
  quality?: number;
  page?: number;
  timeSeconds?: number;
  signal?: AbortSignal;
}

/** 预览结果；内存产物，单次上限 8 MiB。 */
interface IFilePreviewResult {
  data: Uint8Array;
  mimeType: 'image/png' | 'image/jpeg' | 'image/webp';
  width: number;
  height: number;
  page?: number;
  timeSeconds?: number;
}

/** 转换输入。 */
interface IFileConvertInput {
  target: {
    /** 目标 MIME（IANA 注册或 vendor 类型）。 */
    mimeType: string;
    /** 建议输出扩展名；省略时按 MIME 决定。 */
    extension?: string;
  };
  output: IFileProcessTarget;
  options?: IFileProcessFormatOptions;
  signal?: AbortSignal;
}

/** 编排步骤（诊断用途）。 */
interface IFileProcessPipelineStep {
  /** 内容处理服务名或平台接口名。 */
  service: string;
  /** 原子操作名（服务方法名或平台动作名）。 */
  operation: string;
}

/** 转换结果。 */
interface IFileConvertResult {
  outputUri: string;
  mimeType: string;
  size: number;
  /** 最终产物字节的 sha256:<hex>。 */
  digest: string;
  /** 实际执行步骤，按顺序。 */
  pipeline: IFileProcessPipelineStep[];
  warnings?: Array<{ code: string; message: string }>;
}

/** 另存输入。 */
interface IFileSaveAsInput {
  output: IFileProcessTarget;
  signal?: AbortSignal;
}

/** 另存结果。 */
interface IFileSaveAsResult {
  outputUri: string;
  size: number;
  digest: string;
}

/** 比较算法。 */
type FileCompareAlgorithm = 'hash' | 'binary' | 'semantic';

/** 差异块；范围坐标按各侧来源自身计算，hash / binary 不含内容文本。 */
interface IFileDiffHunk {
  type: 'add' | 'remove' | 'modify';
  leftRange?: { start: number; end: number };
  rightRange?: { start: number; end: number };
  /** 行文本；仅 semantic 提供，不携带来源之外的上下文。 */
  leftContent?: string;
  rightContent?: string;
}

/** 比较选项；与算法不匹配的字段不生效（不报错）。 */
interface IFileCompareOptions {
  /** unified diff 上下文行数，0-10，默认 3；仅 semantic。 */
  context?: number;
  /** 忽略行内空白与行首行尾空白；仅 semantic。 */
  ignoreWhitespace?: boolean;
  /** 忽略大小写；仅 semantic。 */
  ignoreCase?: boolean;
  /** 差异块上限，1-4096，默认 512；超出截断并置 truncated。 */
  maxHunks?: number;
  /** 附加 unified diff 文本；仅 semantic。 */
  unified?: boolean;
}

/** 比较输入。 */
interface IFileCompareInput {
  /** 对侧来源（受管引用）；本会话来源为左/本侧。 */
  other: string;
  algorithm?: FileCompareAlgorithm;
  options?: IFileCompareOptions;
  signal?: AbortSignal;
}

/** 比较结果。 */
interface IFileCompareResult {
  identical: boolean;
  algorithm: FileCompareAlgorithm;
  left: IFileProcessingSourceInfo;
  right: IFileProcessingSourceInfo;
  /** 差异块；identical 时为空数组。 */
  diffs: IFileDiffHunk[];
  /** 差异块是否因 maxHunks 截断。 */
  truncated: boolean;
  stats: { additions: number; deletions: number; modifications: number };
  /** 两侧内容摘要（sha256:<hex>）；hash 算法时返回。 */
  leftDigest?: string;
  rightDigest?: string;
  /** unified diff 文本；仅 semantic 且 unified: true 时存在。 */
  unifiedDiff?: string;
}

/** 文件处理会话。 */
interface IFileProcessingSession {
  /** 不透明会话标识；不持久化、不跨 Fork 传递、不复用。 */
  readonly sessionId: string;
  readonly state: FileProcessingSessionState;
  readonly source: IFileProcessingSourceInfo;
  preview(input?: IFilePreviewInput): Promise<IFilePreviewResult>;
  convert(input: IFileConvertInput): Promise<IFileConvertResult>;
  saveAs(input: IFileSaveAsInput): Promise<IFileSaveAsResult>;
  /** 与本会话来源（左/本侧）比较对侧来源；算法与结果见 §3.10。 */
  compare(input: IFileCompareInput): Promise<IFileCompareResult>;
  /** 订阅进度；返回的 Dispose 撤销订阅。 */
  onProgress(handler: (progress: IFileProcessProgress) => void): Dispose;
  /** 取消在途操作；无在途操作时为 no-op。 */
  cancel(): void;
  /** 关闭会话；幂等。 */
  close(): Promise<void>;
}

/** 文件处理服务。 */
interface IFileProcessingService {
  openSession(input: IFileProcessingOpenInput): Promise<IFileProcessingSession>;
}
```

类型由提供方随插件发布（契约包 `chips.service.file-processing/contracts`）；消费方不得在插件内重复定义同名接口（[04-能力契约规范](./04-能力契约规范.md) §2.1 不变量 3 的同口径约束）。

### 3.2 会话状态机

```text
openSession(source)
  │  归一引用 · 获取句柄 · 探测格式 · 记录版本戳
  ▼
opened ──preview / convert / saveAs / compare──▶ busy ──成功/失败──▶ opened
  │                                     │
  │                                     └──cancel()──▶ opened
  │  源版本戳变化 / 句柄失效
  ▼
stale（拒绝新操作，返回 FILE_PROCESS_SOURCE_CHANGED；只能重新 openSession）
  ▲
  └── close()（任意状态）──▶ closed（终态，幂等）
```

- `opened`：可发起操作；`busy`：单一在途操作，调用方不得并发发起（见 §3.8）。
- `stale` 不是独立 state 取值：`state` 仍为 `opened`，但后续操作确定性地返回 `FILE_PROCESS_SOURCE_CHANGED`，直到重新打开。
- `close()` 在 `busy` 时先取消在途操作再关闭；Fork 释放时服务自动执行同一序列。
- 会话不持久化：服务重启、插件更新、进程崩溃后不存在可恢复会话。

### 3.3 打开

1. 归一 `source` 为受管引用；非法或越权由资源层返回 `RESOURCE_*`，不建立会话。
2. 获取来源句柄（只读、持租约），读取 `IResourceMetadata`（`size` / `mimeType` / `version` / `scope`）。
3. 探测格式：按 MIME 与魔数归一为 `format`；探测与识别由内容处理服务完成，服务自身不解析内容。探测失败时 `format` 为 `unknown`，仍允许建立会话（`convert` 阶段按实际支持情况返回 `FILE_PROCESS_SOURCE_UNSUPPORTED`）。
4. 返回会话；`openSession` 不读取正文，不产生产物。
5. 同一来源可被多个会话绑定；句柄与租约相互独立，写入目标由 WriteCoordinator 串行化（[04-资源管理设计](../01-架构设计/01-子系统设计/04-资源管理设计.md) §4.8）。

约束：

- `.card` / `.box`（`format` 为 `card` / `box`）建立会话仅用于 `convert`；`preview` 返回 `FILE_PROCESS_SOURCE_UNSUPPORTED`（`details.hint` 指向受控封面渲染接口），`saveAs` 返回 `FILE_PROCESS_SOURCE_UNSUPPORTED`（`details.hint` 指向 `document:export`）。
- 打开阶段被取消返回 `FILE_PROCESS_CANCELLED`，不产生半开状态。

### 3.4 预览

- 输入为显示需求（尺寸、适应方式、格式、页码、时间点）；输出为内存字节，单次上限 8 MiB（`platform.resources.maxInlineBytes`，[04-资源管理设计](../01-架构设计/01-子系统设计/04-资源管理设计.md) §十二）。
- 预览是**有界派生结果**：提供方必须按其支持矩阵收敛到请求尺寸；无法在 8 MiB 内表达时返回 `FILE_PROCESS_OUTPUT_TOO_LARGE`，调用方应降低尺寸或改用 `convert` 生成文件产物。
- 预览不写用户工作区、不占用目标作用域；服务可在提供方插件作用域内缓存派生结果，缓存键为 `(来源 URI, 版本戳, 归一化预览选项)`，来源版本戳变化或句柄失效即失效。
- 预览产物不发布事件；同一会话内重复预览命中缓存时同步返回，不推进进度（直接上报 `completed` 100%）。

### 3.5 转换

- `convert` 把来源编排为 `target.mimeType` 的单一文件产物；批量转换由调用方对多个会话/多次调用编排，服务不提供批量入口（批量并发预算由调用方与平台配额控制）。
- 提供方选择：文件处理服务按 `(source.format, target.mimeType)` 在已注入的内容处理服务中选择；选择结果只依赖静态依赖与提供方发布的描述符，不做运行时探测式降级。
- 成功条件：管线全部步骤成功、暂存产物校验通过、受控提交完成、产物摘要可读。
- 结果中的 `pipeline` 只作诊断；调用方不得根据 `pipeline` 分支业务逻辑。
- 支持的格式组合、选项语义与原子步骤数由内容处理服务提供方的契约与支持矩阵定义；本文不复制矩阵，不支持组合返回 `FILE_PROCESS_TARGET_UNSUPPORTED`（`details` 携带 `sourceFormat` / `targetMimeType`）。

### 3.6 另存

- `saveAs` 把来源字节按原格式复制为文件产物：不解析、不解码、不重编码；等价于带进度与取消的流式拷贝。
- 大文件（> 8 MiB）必须流式：来源句柄 `stream()` 分块读取，产物经流式受控提交写出（§五）。
- `saveAs` 的产物与来源字节等价（摘要可比对）；不得在另存路径中夹带任何格式改写。
- 需要格式转换时使用 `convert`；需要改变文档形态时使用 `document:export`。

### 3.7 进度与取消

进度：

- 每次操作上报固定阶段序列；`percent` 在操作内单调不减，到达 `completed` 时为 100。
- `prepare`（0-10）、`transform`（10-80）、`stage`（80-90）、`publish`（90-100）、`cleanup`（清理期间保持 100，不回落）。阶段顺序固定；跳过阶段时百分比直接前进，不出现回落。
- 各操作使用的阶段子集：`convert` / `saveAs` 使用完整序列；`preview` 与 `compare` 使用 `prepare` → `transform` → `completed`，不产生产物阶段。
- 进度回调在服务调用线程同步触发；回调不得阻塞、不得抛异常依赖；回调内耗时处理必须转交 `ctx.task`。

取消：

- 取消入口有两个且语义一致：请求上的 `signal` 与 `session.cancel()`；任一触发即取消在途操作。
- 取消是协作式的：服务在阶段边界与提供方声明的安全点检查取消；内容处理服务必须响应其调用 `signal`（[04-能力契约规范](./04-能力契约规范.md) §2.1 不变量 4）。
- 取消结果：在途 Promise 以 `FILE_PROCESS_CANCELLED` 拒绝（`retryable: true`）；暂存产物清理；目标保持原状（未提交或原子替换未发生）；会话回到 `opened`。
- `publish` 阶段进入提交点后不可取消：此时取消请求被忽略，操作按提交结果收口（成功或 `FILE_PROCESS_WRITE_FAILED`）。
- 会话关闭、Fork 释放、服务代际切换都会取消在途操作，未提交产物一律清理。

### 3.8 并发与源变更

- 同一会话同一时间只允许一个在途操作；`busy` 期间发起新操作返回 `FILE_PROCESS_BUSY`（`retryable: true`）。
- 不同会话相互独立；同一目标路径的写入由 WriteCoordinator 串行化，后到提交在前者完成后执行；目标存在性判定在提交时最终确认。
- 来源版本戳在每次操作前与提交前复查；不一致时操作失败并返回 `FILE_PROCESS_SOURCE_CHANGED`（`retryable: false`），会话进入 stale，只能重新打开。
- 句柄 `invalidated`（来源变化、权限撤销、作用域注销）同样触发上述行为；服务不得在失效句柄上继续消费字节（禁止"模型旧 + 字节新"，[04-资源管理设计](../01-架构设计/01-子系统设计/04-资源管理设计.md) §4.5）。

### 3.9 关闭与释放

- `close()`：取消在途操作 → 清空暂存区 → 释放来源句柄与订阅 → 会话置 `closed`；幂等，重复调用为 no-op。
- 关闭后调用任何操作返回 `FILE_PROCESS_SESSION_CLOSED`；`sessionId` 不复用。
- 服务不持有可跨会话复用的字节缓存；可重算缓存只存在于提供方插件作用域并随其缓存策略治理。

### 3.10 比较与差异

比较（`compare`）对**会话来源（左/本侧）**与**对侧来源（`other`）**做一次性差异计算：不产生产物、不写文件、不改变任一侧来源。对侧来源在单次操作内建立只读句柄并在结束时释放；句柄获取、作用域与权限判定与会话来源同口径，任一来源越权由资源层返回 `RESOURCE_*`，不执行比较。

算法（`algorithm`，默认 `hash`）：

| 算法 | 语义 | 适用来源 | 输出 |
|---|---|---|---|
| `hash` | 内容摘要相等性：流式计算两侧 SHA-256，只判定是否一致 | 任意普通文件来源 | `leftDigest` / `rightDigest`；不一致时为覆盖全长的单个 `modify` 块 |
| `binary` | 字节级差异：流式逐块比较，连续差异折叠为差异块 | 任意普通文件来源 | 差异块（只有字节范围，无内容文本） |
| `semantic` | 文本行级语义对比：按行拆分、按选项归一化后求差异块 | 仅文本类来源 | 差异块（含行文本）；`unified: true` 时附 unified diff 文本 |

类型见 §3.1（`IFileCompareInput` / `IFileCompareOptions` / `IFileDiffHunk` / `IFileCompareResult`）。

规则：

1. **不解析内容**：比较只做字节运算与文本行级运算；禁止解包或解释任何结构化容器与格式（ZIP、OLE、YAML、JSON 结构、`.card` / `.box`、Office 容器）。`semantic` 只做文本解码（UTF-8，含 BOM 处理）与行拆分，行拆分对 `\r\n` 与 `\n` 等价。
2. **文本类来源判定**：`semantic` 仅接受文本类来源（MIME `text/*`，或按内容处理服务探测归一为文本类的格式，如 `text/plain`、`text/markdown`、`application/json`、`application/yaml`、`application/xml` 与源码 / 配置类文本）；其他来源确定性返回 `FILE_PROCESS_SOURCE_UNSUPPORTED`（`details.hint` 指向 [04-能力契约规范](./04-能力契约规范.md) §八）。`.card` / `.box` 只允许 `hash` 与 `binary`。
3. **格式语义对比不在本契约**：按格式语义对比（如 Office 文档正文、PDF 内容）必须先经内容处理服务提取可比内容；当前登记不提供比较类原子能力，不支持组合按 `FILE_PROCESS_SOURCE_UNSUPPORTED` 收口，禁止在文件处理服务内实现第二套格式解析。
4. **无历史版本**：平台不提供文档版本历史与持久备份版本；两版本对比 = 对两个来源分别取受管引用后调用 `compare`。旧 `file.compare` / `file.versionDiff` 链路与 `.bak` 备份版本语义废止。
5. **流式与上限**：两侧读取遵守 §五：`hash` 与 `binary` 必须流式，> 8 MiB 来源禁止整体载入内存；`semantic` 需要完整行序列，来源大小上限 8 MiB，超过返回 `FILE_PROCESS_INPUT_TOO_LARGE`（`details.hint` 指向 `binary` / 分段对比）。差异块数超过 `maxHunks` 时截断并置 `truncated: true`；`unified` 文本超过 8 MiB（`platform.resources.maxInlineBytes`）时返回 `FILE_PROCESS_OUTPUT_TOO_LARGE`。
6. **进度与取消**：阶段为 `prepare`（获取对侧句柄、复查版本戳）→ `transform`（读取与计算）→ `completed`；`signal` 与 `session.cancel()` 语义同 §3.7，取消返回 `FILE_PROCESS_CANCELLED`。
7. **源变更**：两侧来源版本戳在操作前与结果返回前复查；**会话来源**变化按 §3.8 使会话进入 stale；**对侧来源**变化时本次操作失败返回 `FILE_PROCESS_SOURCE_CHANGED`（`retryable: false`），会话保持 `opened`，可重新发起比较。
8. **失败归一**：读取、解码或计算失败归一为 `FILE_PROCESS_COMPARE_FAILED`（原始错误进 `details.cause`）；资源与权限错误保持 `RESOURCE_*` 原码不包装。
