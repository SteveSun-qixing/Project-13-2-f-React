## 六、完整示例

### 6.1 请求-响应

```typescript
interface IFileListInput {
  directory: string;
}

interface IFileListOutput {
  entries: string[];
}

interface IFileReadInput {
  path: string;
  encoding?: 'utf-8' | 'base64';
}

interface IFileReadOutput {
  content: string;
  encoding: 'utf-8' | 'base64';
}

/**
 * 读取卡片包目录中的第一个文本条目。
 *
 * 能力输入 schema 由《04-能力契约规范》固化；上述接口是调用方的编译期类型声明。
 */
export async function readFirstEntry(directory: string): Promise<string> {
  const listing = await window.chips.request<IFileListOutput>('file:list', {
    directory,
  } satisfies IFileListInput);

  const entry = listing.entries.at(0);
  if (entry === undefined) {
    throw new Error(`目录为空：${directory}`);
  }

  const file = await window.chips.request<IFileReadOutput>('file:read', {
    path: entry,
    encoding: 'utf-8',
  } satisfies IFileReadInput);

  return file.content;
}
```

### 6.2 事件订阅

```typescript
import { useEffect } from 'react';
import type { ISubscription } from '@chips/sdk';

interface IFileChangedPayload {
  uri: string;
  kind: 'created' | 'modified' | 'deleted' | 'renamed';
  previousUri?: string;
}

function isFileChangedPayload(payload: unknown): payload is IFileChangedPayload {
  if (typeof payload !== 'object' || payload === null) {
    return false;
  }
  const candidate = payload as Record<string, unknown>;
  return (
    typeof candidate.uri === 'string' &&
    (candidate.kind === 'created' ||
      candidate.kind === 'modified' ||
      candidate.kind === 'deleted' ||
      candidate.kind === 'renamed')
  );
}

/**
 * 在视图中监听指定文件的变更；订阅随组件卸载自动退订。
 */
export function useFileChanged(
  uri: string,
  onChanged: (payload: IFileChangedPayload) => void,
): void {
  useEffect(() => {
    const subscription: ISubscription = window.chips.subscribe(
      'file:changed',
      (payload: unknown) => {
        if (isFileChangedPayload(payload) && payload.uri === uri) {
          onChanged(payload);
        }
      },
    );

    return () => {
      subscription.dispose();
    };
  }, [uri, onChanged]);
}
```

非组件场景必须持有句柄并显式释放：

```typescript
const subscription = window.chips.subscribe('document:saved', (payload: unknown) => {
  // 按事件契约收窄 payload 后处理
});

// 监听结束时（视图卸载、插件停用、任务完成）
subscription.dispose();
```

### 6.3 资源访问

```typescript
import { useEffect, useState } from 'react';
import { ChipsError } from '@chips/sdk';
import type { IResourceMetadata } from '@chips/sdk';

/**
 * 读取资源元数据，并在资源失效时清空结果。
 *
 * 字节加载交给渲染引擎：<img src={uri}> 由 chips-resource:// 协议处理器按 Range 提供。
 */
export function useResourceMetadata(uri: string): IResourceMetadata | null {
  const [metadata, setMetadata] = useState<IResourceMetadata | null>(null);

  useEffect(() => {
    const handle = window.chips.resource(uri);
    const invalidation = handle.onInvalidate((reason) => {
      if (reason !== 'source-changed') {
        setMetadata(null);
      }
    });

    let cancelled = false;
    handle
      .metadata()
      .then((value) => {
        if (!cancelled) {
          setMetadata(value);
        }
      })
      .catch((error: unknown) => {
        if (!(error instanceof ChipsError) || error.code !== 'RESOURCE_HANDLE_INVALIDATED') {
          console.error('读取资源元数据失败', error);
        }
      });

    return () => {
      cancelled = true;
      invalidation.dispose();
      handle.release();
    };
  }, [uri]);

  return metadata;
}
```

小资源直接读取字节：

```typescript
async function readSmallResource(uri: string): Promise<Uint8Array> {
  const handle = window.chips.resource(uri);
  try {
    return await handle.read();
  } finally {
    handle.release();
  }
}
```

### 6.4 错误处理

```typescript
import { ChipsError } from '@chips/sdk';

interface IFileWriteInput {
  path: string;
  data: string;
}

export type SaveOutcome = 'saved' | 'permission-denied' | 'retry-later';

/**
 * 保存文本文件；按错误码分支处理，禁止匹配 message 文本。
 *
 * BRIDGE_PERMISSION_DENIED 的 details 结构见《02-Bridge三层设计》5.4 节。
 */
export async function saveTextFile(path: string, text: string): Promise<SaveOutcome> {
  try {
    await window.chips.request<void>('file:write', {
      path,
      data: text,
    } satisfies IFileWriteInput);
    return 'saved';
  } catch (error) {
    if (!(error instanceof ChipsError)) {
      throw error;
    }
    switch (error.code) {
      case 'BRIDGE_PERMISSION_DENIED':
        return 'permission-denied';
      case 'BRIDGE_TIMEOUT':
      case 'BRIDGE_SESSION_BUSY':
      case 'BRIDGE_QUEUE_OVERFLOW':
        return 'retry-later';
      default:
        throw error;
    }
  }
}
```
