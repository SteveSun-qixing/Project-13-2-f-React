### 9.6 `src/render.ts`（渲染方法）

```typescript
import type {
  BaseCardChange,
  BaseCardView,
  Dispose,
  EditorState,
  IBaseCardEditor,
  IBaseCardEditorEventMap,
  ICardPatchOperation,
  ISubscription,
} from '@chips/sdk';
import { ChipsError } from '@chips/sdk';

const DEFAULT_FONT_SIZE = 16;
const DEFAULT_LINE_HEIGHT = 1.6;
const MIN_FONT_SIZE = 12;
const MAX_FONT_SIZE = 48;

interface IRichTextConfig {
  fontSize: number;
  lineHeight: number;
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function readText(content: Record<string, unknown>): string {
  return typeof content.content_text === 'string' ? content.content_text : '';
}

function readConfig(content: Record<string, unknown>): IRichTextConfig {
  const raw = isRecord(content.config) ? content.config : {};
  return {
    fontSize: typeof raw.fontSize === 'number' ? raw.fontSize : DEFAULT_FONT_SIZE,
    lineHeight: typeof raw.lineHeight === 'number' ? raw.lineHeight : DEFAULT_LINE_HEIGHT,
  };
}

function escapeHtml(text: string): string {
  return text
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;');
}

function applyOperation(
  target: Record<string, unknown>,
  keys: readonly string[],
  operation: ICardPatchOperation,
): void {
  const head = keys[0];
  if (head === undefined) {
    return;
  }
  if (keys.length === 1) {
    if (operation.op === 'remove') {
      delete target[head];
    } else {
      target[head] = operation.value;
    }
    return;
  }
  const child = target[head];
  const nested = isRecord(child) ? { ...child } : {};
  target[head] = nested;
  applyOperation(nested, keys.slice(1), operation);
}

function applyPatch(
  content: Record<string, unknown>,
  operations: readonly ICardPatchOperation[],
): Record<string, unknown> {
  const next: Record<string, unknown> = { ...content };
  for (const operation of operations) {
    const keys = operation.path.split('/').filter((key) => key.length > 0);
    if (keys.length > 0) {
      applyOperation(next, keys, operation);
    }
  }
  return next;
}

function applyChange(
  base: BaseCardView,
  change: BaseCardChange,
  revision: number,
): BaseCardView {
  const content =
    change.kind === 'replace' ? change.content : applyPatch(base.content, change.operations);
  return { ...base, revision, content };
}

export function renderView(basecard: BaseCardView, container: HTMLElement): Dispose {
  const config = readConfig(basecard.content);
  const article = document.createElement('article');
  article.style.fontSize = `${config.fontSize}px`;
  article.style.lineHeight = String(config.lineHeight);
  article.textContent = readText(basecard.content);
  container.replaceChildren(article);

  return () => {
    article.remove();
  };
}

class RichTextEditor implements IBaseCardEditor {
  public state: EditorState = 'active';

  private base: BaseCardView;
  private revision: number;
  private pending: BaseCardChange | null = null;
  private isDirtyFlag = false;
  private readonly root: HTMLDivElement;
  private readonly textarea: HTMLTextAreaElement;
  private readonly fontSizeInput: HTMLInputElement;
  private readonly changeHandlers = new Set<(payload: IBaseCardEditorEventMap['change']) => void>();
  private readonly dirtyHandlers = new Set<(payload: IBaseCardEditorEventMap['dirty']) => void>();

  public constructor(basecard: BaseCardView, container: HTMLElement) {
    this.base = basecard;
    this.revision = basecard.revision;

    this.root = document.createElement('div');

    this.textarea = document.createElement('textarea');
    this.textarea.rows = 12;

    this.fontSizeInput = document.createElement('input');
    this.fontSizeInput.type = 'number';
    this.fontSizeInput.min = String(MIN_FONT_SIZE);
    this.fontSizeInput.max = String(MAX_FONT_SIZE);

    this.root.append(this.textarea, this.fontSizeInput);
    container.replaceChildren(this.root);

    this.renderFromView(basecard);
    this.textarea.addEventListener('input', this.handleInput);
    this.fontSizeInput.addEventListener('input', this.handleInput);
  }

  public isDirty(): boolean {
    return this.pending !== null;
  }

  public getChange(): BaseCardChange | null {
    return this.pending;
  }

  public markCommitted(revision: number): void {
    if (this.state === 'disposed' || revision <= this.revision) {
      return;
    }
    const committed = this.pending;
    this.revision = revision;
    if (committed !== null) {
      this.base = applyChange(this.base, committed, revision);
    }
    if (this.pending === committed) {
      this.pending = null;
      this.setDirty(false);
    }
  }

  public update(basecard: BaseCardView, revision: number): void {
    if (this.state === 'disposed' || revision <= this.revision) {
      return;
    }
    this.revision = revision;
    this.base = { ...basecard, revision };
    this.pending = null;
    this.setDirty(false);
    this.renderFromView(basecard);
  }

  public subscribe<TEvent extends keyof IBaseCardEditorEventMap>(
    event: TEvent,
    handler: (payload: IBaseCardEditorEventMap[TEvent]) => void,
  ): ISubscription {
    if (this.state === 'disposed') {
      return { dispose: (): void => undefined };
    }
    if (event === 'change') {
      const typed = handler as (payload: IBaseCardEditorEventMap['change']) => void;
      this.changeHandlers.add(typed);
      return {
        dispose: (): void => {
          this.changeHandlers.delete(typed);
        },
      };
    }
    const typed = handler as (payload: IBaseCardEditorEventMap['dirty']) => void;
    this.dirtyHandlers.add(typed);
    return {
      dispose: (): void => {
        this.dirtyHandlers.delete(typed);
      },
    };
  }

  public focus(): void {
    if (this.state === 'active') {
      this.textarea.focus();
    }
  }

  public dispose(): void {
    if (this.state === 'disposed') {
      return;
    }
    this.textarea.removeEventListener('input', this.handleInput);
    this.fontSizeInput.removeEventListener('input', this.handleInput);
    this.root.remove();
    this.changeHandlers.clear();
    this.dirtyHandlers.clear();
    this.pending = null;
    this.setDirty(false);
    this.state = 'disposed';
  }

  private readonly handleInput = (): void => {
    if (this.state === 'disposed') {
      return;
    }
    const fontSize = Number(this.fontSizeInput.value);
    const content: Record<string, unknown> = {
      ...this.base.content,
      content_text: this.textarea.value,
      config: {
        ...readConfig(this.base.content),
        fontSize:
          Number.isFinite(fontSize) && fontSize >= MIN_FONT_SIZE && fontSize <= MAX_FONT_SIZE
            ? fontSize
            : DEFAULT_FONT_SIZE,
      },
    };
    const change: BaseCardChange = { kind: 'replace', content };
    this.pending = change;
    for (const handler of this.changeHandlers) {
      handler({ change });
    }
    this.setDirty(true);
  };

  private renderFromView(basecard: BaseCardView): void {
    const config = readConfig(basecard.content);
    this.textarea.value = readText(basecard.content);
    this.fontSizeInput.value = String(config.fontSize);
  }

  private setDirty(isDirty: boolean): void {
    if (this.isDirtyFlag === isDirty) {
      return;
    }
    this.isDirtyFlag = isDirty;
    for (const handler of this.dirtyHandlers) {
      handler({ isDirty });
    }
  }
}

export function renderEditor(basecard: BaseCardView, container: HTMLElement): IBaseCardEditor {
  return new RichTextEditor(basecard, container);
}

export async function exportCard(basecard: BaseCardView, format: string): Promise<Blob> {
  if (format !== 'html') {
    throw new ChipsError({
      code: 'CARD_EXPORT_UNSUPPORTED',
      message: `该卡片类型不支持导出为 ${format}`,
      details: { format },
      retryable: false,
    });
  }
  const config = readConfig(basecard.content);
  const body = escapeHtml(readText(basecard.content)).replaceAll('\n', '<br>');
  const html =
    `<article style="font-size: ${config.fontSize}px; line-height: ${config.lineHeight}">` +
    `${body}</article>`;
  return new Blob([html], { type: 'text/html' });
}
```
