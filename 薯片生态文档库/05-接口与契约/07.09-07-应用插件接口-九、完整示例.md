## 九、完整示例

示例为一个最小应用插件 `chips.example.text-viewer`：窗口应用，注册启动器与 `.txt` / `.md` 文件关联，读取自身配置控制标题，支持 `text-viewer:reset-title` 与 `text-viewer:open-uri` 两个命令，并在退出前释放订阅。主逻辑位于 `src/index.ts`（构建为 `dist/index.js`），UI 位于 `src/ui/AppView.tsx`（构建为 `render.js`，由平台窗口壳装载）。

### 9.3 主逻辑 src/index.ts

```typescript
import type {
  IAppConfigChange,
  IAppContext,
  IAppPlugin,
  IEventSubscription,
} from '@chips/sdk';

const PLUGIN_ID = 'chips.example.text-viewer';
const TITLE_KEY = `plugins.${PLUGIN_ID}.window-title`;
const DEFAULT_TITLE = '文本查看器';

class AppInvokeError extends Error {
  public readonly code: string;
  public readonly details?: unknown;

  public constructor(code: string, message: string, details?: unknown) {
    super(message);
    this.name = 'AppInvokeError';
    this.code = code;
    this.details = details;
  }
}

const isSupportedUri = (uri: string): boolean =>
  uri.startsWith('file://') || uri.startsWith('chips-resource://');

const fileNameOf = (uri: string): string => {
  const trimmed = uri.endsWith('/') ? uri.slice(0, -1) : uri;
  const slashIndex = trimmed.lastIndexOf('/');
  return slashIndex >= 0 ? trimmed.slice(slashIndex + 1) : trimmed;
};

const readUriArg = (args: unknown): string => {
  if (typeof args === 'object' && args !== null && 'uri' in args) {
    const candidate = args as { uri?: unknown };
    if (typeof candidate.uri === 'string') {
      return candidate.uri;
    }
  }
  throw new AppInvokeError(
    'APP_COMMAND_ARGS_INVALID',
    '命令参数必须包含字符串 uri 字段',
  );
};

class TextViewerApp implements IAppPlugin {
  private current: IAppContext | null = null;
  private titleSubscription: IEventSubscription | null = null;

  public async launch(ctx: IAppContext): Promise<void> {
    this.current = ctx;

    let configured: string | null = null;
    try {
      const view = await ctx.config.get('window-title');
      configured = typeof view.value === 'string' ? view.value : null;
    } catch {
      // 自身配置读取失败不阻断启动，回退默认标题；平台统一记录诊断
      configured = null;
    }
    await this.applyTitle(configured ?? DEFAULT_TITLE);

    this.titleSubscription = ctx.on<IAppConfigChange>('config:changed', (payload) => {
      if (payload.key !== TITLE_KEY) {
        return;
      }
      if (typeof payload.effectiveValue === 'string') {
        void this.applyTitle(payload.effectiveValue);
      }
    });

    ctx.lifecycle.beforeExit(async () => {
      this.titleSubscription?.dispose();
      this.titleSubscription = null;
      this.current = null;
    });
  }

  public async handleCommand(command: string, args: unknown): Promise<void> {
    if (command === 'text-viewer:reset-title') {
      await this.applyTitle(DEFAULT_TITLE);
      return;
    }
    if (command === 'text-viewer:open-uri') {
      await this.openFile(readUriArg(args));
      return;
    }
    throw new AppInvokeError(
      'APP_COMMAND_UNSUPPORTED',
      `不支持的命令：${command}`,
      { command },
    );
  }

  public async openFile(uri: string): Promise<void> {
    if (!isSupportedUri(uri)) {
      throw new AppInvokeError(
        'APP_URI_INVALID',
        `只接受 file:// 或 chips-resource:// URI：${uri}`,
        { uri },
      );
    }
    await this.applyTitle(`${DEFAULT_TITLE} - ${fileNameOf(uri)}`);
  }

  private async applyTitle(title: string): Promise<void> {
    const window = this.current?.window ?? null;
    if (window === null) {
      return;
    }
    await window.setTitle(title);
  }
}

export default new TextViewerApp();
```

### 9.4 UI src/ui/AppView.tsx

```tsx
import React from 'react';
import { Button, Empty, Spin } from '@chips/ui';
import { chips } from '@chips/sdk';
import type { ISubscription } from '@chips/sdk';

const NAMESPACE = 'plugins.chips.example.text-viewer';
const TITLE_KEY = 'window-title';
const DEFAULT_TITLE = '文本查看器';

interface IConfigGetResult {
  key: string;
  value: unknown;
  layer: 'platform' | 'user' | 'workspace' | 'plugin';
  isDefault: boolean;
}

interface IConfigChangedPayload {
  key: string;
  effectiveValue: unknown;
}

const isConfigChangedPayload = (payload: unknown): payload is IConfigChangedPayload => {
  if (typeof payload !== 'object' || payload === null) {
    return false;
  }
  const candidate = payload as { key?: unknown };
  return typeof candidate.key === 'string' && 'effectiveValue' in candidate;
};

type LoadState = 'loading' | 'ready' | 'error';

export const AppView: React.FC = () => {
  const [title, setTitle] = React.useState<string>(DEFAULT_TITLE);
  const [state, setState] = React.useState<LoadState>('loading');

  React.useEffect(() => {
    let active = true;

    const load = async (): Promise<void> => {
      try {
        const result = await chips.request<IConfigGetResult>('config:get', {
          namespace: NAMESPACE,
          key: TITLE_KEY,
        });
        if (!active) {
          return;
        }
        if (typeof result.value === 'string') {
          setTitle(result.value);
        }
        setState('ready');
      } catch {
        if (active) {
          setState('error');
        }
      }
    };

    void load();

    const subscription: ISubscription = chips.subscribe('config:changed', (payload) => {
      if (!isConfigChangedPayload(payload) || payload.key !== `${NAMESPACE}.${TITLE_KEY}`) {
        return;
      }
      if (typeof payload.effectiveValue === 'string') {
        setTitle(payload.effectiveValue);
      }
    });

    return () => {
      active = false;
      subscription.dispose();
    };
  }, []);

  const resetTitle = React.useCallback((): void => {
    void chips.request('config:set', {
      namespace: NAMESPACE,
      key: TITLE_KEY,
      value: DEFAULT_TITLE,
      scope: 'user',
    });
  }, []);

  if (state === 'loading') {
    return <Spin />;
  }

  if (state === 'error') {
    return <Empty description="读取应用配置失败" />;
  }

  return (
    <div>
      <div>{title}</div>
      <Button onClick={resetTitle}>恢复默认标题</Button>
    </div>
  );
};
```
