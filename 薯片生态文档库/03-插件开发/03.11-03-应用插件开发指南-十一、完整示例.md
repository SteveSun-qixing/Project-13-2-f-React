## 十一、完整示例

示例为一个最小应用插件 `chips.example.quick-notes`（随手记）：窗口应用，注册启动器与 `.txt` / `.md` 文件关联，读取自身配置控制窗口标题，UI 提供条目输入、删除、复制全部与条目数开关。

### 11.1 manifest.yaml

```yaml
id: chips.example.quick-notes
version: 1.0.0
type: app
name: 随手记
description: 随手记录并管理简短事项
author: 薯片示例
license: MIT
runtime:
  targets: [host]
  background: false
permissions:
  - clipboard:access
fileAssociations:
  - extensions: [.txt, .md]
    action: open
ui:
  surface: window
  launcher:
    icon: assets/icons/app.svg
    name: 随手记
    description: 随手记录并管理简短事项
```

启动器图标文件 `assets/icons/app.svg` 随包提供（内容省略）。

### 11.2 config.schema.json

```json
{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "title": "随手记配置",
  "x-chips-schema-version": 1,
  "type": "object",
  "additionalProperties": false,
  "properties": {
    "title": {
      "type": "string",
      "default": "随手记",
      "title": "窗口标题"

      "default": true,
      "title": "显示条目数"
    }
  }

```

配置键字符集为小写字母、数字与连字符；默认值不落盘，只有显式覆盖才写入 `user` 或 `workspace` 层（[10-配置系统设计](../01-架构设计/01-子系统设计/10-配置系统设计.md) §4）。

### 11.3 主逻辑 src/index.ts

```typescript
import type { IAppContext, IAppPlugin, IEventSubscription } from '@chips/sdk';

const CONFIG_KEY = 'title';
const DEFAULT_TITLE = '随手记';

class QuickNotesError extends Error {
  public readonly code: string;
  public readonly details?: unknown;

  public constructor(code: string, message: string, details?: unknown) {
    super(message);
    this.name = 'QuickNotesError';
    this.code = code;
    this.details = details;
  }
}

const isSupportedUri = (uri: string): boolean =>
  uri.startsWith('file://') || uri.startsWith('chips-resource://'); // file:// 仅目录态：平台在投递前已归一为受管作用域 URI

const fileNameOf = (uri: string): string => {
  const trimmed = uri.endsWith('/') ? uri.slice(0, -1) : uri;
  const separator = trimmed.lastIndexOf('/');
  return separator >= 0 ? trimmed.slice(separator + 1) : trimmed;
};

interface ISetTitleArgs {
  readonly title: string;
}

const parseSetTitleArgs = (args: unknown): ISetTitleArgs => {
  if (typeof args === 'object' && args !== null && 'title' in args) {
    const candidate = args as { title?: unknown };
    if (typeof candidate.title === 'string' && candidate.title.trim().length > 0) {
      return { title: candidate.title };
    }
  }
  throw new QuickNotesError('APP_COMMAND_ARGS_INVALID', '命令参数必须包含非空字符串 title 字段');
};

class QuickNotesApp implements IAppPlugin {
  private context: IAppContext | null = null;
  private titleSubscription: IEventSubscription | null = null;

  public async launch(ctx: IAppContext): Promise<void> {
    this.context = ctx;

    this.titleSubscription = ctx.config.watch(CONFIG_KEY, (change) => {
      if (typeof change.effectiveValue === 'string') {
        void this.applyTitle(change.effectiveValue);
      }
    });

    await this.applyConfiguredTitle();

    if (ctx.args.source === 'file-association') {
      for (const uri of ctx.args.files) {
        await this.openFile(uri);
      }
    }

    if (ctx.args.source === 'command' && ctx.args.command !== undefined) {
      await this.handleCommand(ctx.args.command, ctx.args.commandArgs);
    }

    ctx.lifecycle.beforeExit(() => {
      this.titleSubscription?.dispose();
      this.titleSubscription = null;
      this.context = null;
    });
  }

  public async handleCommand(command: string, args: unknown): Promise<void> {
    switch (command) {
      case 'quick-notes:set-title': {
        const parsed = parseSetTitleArgs(args);
        await this.applyTitle(parsed.title);
        return;
      }
      case 'quick-notes:reset-title': {
        await this.applyConfiguredTitle();
        return;
      }
      case 'quick-notes:close': {
        await this.context?.exit();
        return;
      }
      default: {
        throw new QuickNotesError('APP_COMMAND_UNSUPPORTED', `不支持的命令：${command}`, {
          command,
        });
      }
    }
  }

  public async openFile(uri: string): Promise<void> {
    if (!isSupportedUri(uri)) {
      throw new QuickNotesError(
        'APP_URI_INVALID',
        `只接受受管 file://（目录态、投递前已归一）或 chips-resource:// URI：${uri}`,
        { uri },
      );
    }
    await this.applyTitle(`${DEFAULT_TITLE} - ${fileNameOf(uri)}`);
  }

  private async applyConfiguredTitle(): Promise<void> {
    let title = DEFAULT_TITLE;
    try {
      const view = await this.context?.config.get(CONFIG_KEY);
      if (view !== undefined && typeof view.value === 'string') {
        title = view.value;
      }
    } catch {
      // 读取失败回退默认标题，不阻断启动
    }
    await this.applyTitle(title);
  }

  private async applyTitle(title: string): Promise<void> {
    const appWindow = this.context?.window ?? null;
    if (appWindow === null) {
      return;
    }
    await appWindow.setTitle(title);
  }
}

export default new QuickNotesApp();
```

### 11.4 界面 src/ui/AppView.tsx

```tsx
import React from 'react';
import { Button, Empty, Input, Spin, theme } from '@chips/ui';
import { chips } from '@chips/sdk';
import type {
  CapabilityInput,
  CapabilityOutput,
  IAppConfigChange,
  ISubscription,
} from '@chips/sdk';

const NAMESPACE = 'plugins.chips.example.quick-notes';
const SHOW_COUNT_KEY = 'show-count';
const SHOW_COUNT_FULL_KEY = `${NAMESPACE}.${SHOW_COUNT_KEY}`;

interface INote {
  readonly id: number;
  readonly text: string;
}

type ViewState = 'loading' | 'ready' | 'error';

const isShowCountChange = (payload: unknown): payload is IAppConfigChange => {
  if (typeof payload !== 'object' || payload === null) {
    return false;
  }
  const candidate = payload as { key?: unknown };
  return candidate.key === SHOW_COUNT_FULL_KEY;
};

export const AppView: React.FC = () => {
  const { token } = theme.useToken();
  const nextId = React.useRef(0);
  const [draft, setDraft] = React.useState('');
  const [notes, setNotes] = React.useState<readonly INote[]>([]);
  const [showCount, setShowCount] = React.useState(true);
  const [feedback, setFeedback] = React.useState<string | null>(null);
  const [state, setState] = React.useState<ViewState>('loading');

  React.useEffect(() => {
    let active = true;

    const load = async (): Promise<void> => {
      try {
        const input: CapabilityInput<'config:get'> = {
          namespace: NAMESPACE,
          key: SHOW_COUNT_KEY,
        };
        const view: CapabilityOutput<'config:get'> = await chips.request<
          CapabilityOutput<'config:get'>
        >('config:get', input);
        if (!active) {
          return;
        }
        setShowCount(view.value !== false);
        setState('ready');
      } catch {
        if (active) {
          setState('error');
        }
      }
    };

    void load();

    const subscription: ISubscription = chips.subscribe('config:changed', (payload) => {
      if (isShowCountChange(payload) && typeof payload.effectiveValue === 'boolean') {
        setShowCount(payload.effectiveValue);
      }
    });

    return () => {
      active = false;
      subscription.dispose();
    };
  }, []);

  const addNote = React.useCallback((): void => {
    const text = draft.trim();
    if (text.length === 0) {
      return;
    }
    nextId.current += 1;
    setNotes((previous) => [...previous, { id: nextId.current, text }]);
    setDraft('');
  }, [draft]);

  const removeNote = React.useCallback((id: number): void => {
    setNotes((previous) => previous.filter((note) => note.id !== id));
  }, []);

  const copyAll = React.useCallback((): void => {
    const input: CapabilityInput<'clipboard:write'> = {
      text: notes.map((note) => note.text).join('\n'),
    };
    chips
      .request<void>('clipboard:write', input)
      .then(() => setFeedback('已复制全部内容'))
      .catch(() => setFeedback('复制失败，请检查剪贴板权限'));
  }, [notes]);

  const toggleShowCount = React.useCallback((): void => {
    const next = !showCount;
    setShowCount(next);
    const input: CapabilityInput<'config:set'> = {
      namespace: NAMESPACE,
      key: SHOW_COUNT_KEY,
      value: next,
      scope: 'user',
    };
    chips
      .request<void>('config:set', input)
      .catch(() => setFeedback('配置写入失败'));
  }, [showCount]);

  if (state === 'loading') {
    return <Spin />;
  }

  if (state === 'error') {
    return <Empty description="读取应用配置失败" />;
  }

  return (
    <section
      style={{
        display: 'flex',
        flexDirection: 'column',
        gap: token.marginSM,
        padding: token.padding,
      }}
    >
      <div style={{ display: 'flex', gap: token.marginSM }}>
        <Input
          value={draft}
          onChange={(event) => setDraft(event.target.value)}
          placeholder="输入待记录的内容"
        />
        <Button type="primary" onClick={addNote}>
          添加
        </Button>
        <Button onClick={copyAll} disabled={notes.length === 0}>
          复制全部
        </Button>
        <Button onClick={toggleShowCount}>{showCount ? '隐藏条目数' : '显示条目数'}</Button>
      </div>
      {feedback !== null ? (
        <div style={{ color: token.colorTextSecondary }}>{feedback}</div>
      ) : null}
      {showCount ? (
        <div style={{ color: token.colorTextSecondary }}>{`共 ${notes.length} 条`}</div>
      ) : null}
      {notes.length === 0 ? (
        <Empty description="暂无记录" />
      ) : (
        <ul style={{ margin: 0, paddingLeft: token.paddingLG }}>
          {notes.map((note) => (
            <li key={note.id}>
              <div style={{ display: 'flex', gap: token.marginSM, alignItems: 'center' }}>
                <span>{note.text}</span>
                <Button size="small" onClick={() => removeNote(note.id)}>
                  删除
                </Button>
              </div>
            </li>
          ))}
        </ul>
      )}
    </section>
  );
};

export default AppView;
```

说明：示例把条目保存在窗口内存中，用于演示接口；业务数据持久化应经能力或文档链路写入用户文件。示例不读取文件内容，因此只申请剪贴板权限；需要读取的查看器按第八节申请对应权限。
