  show_tags: boolean;
  information_density: 'compact' | 'comfortable';
};

const DEFAULT_CONFIG: GridConfig = {
  gap: 16,
  show_tags: true,
  information_density: 'comfortable',

const GAP_MIN = 0;
const GAP_MAX = 64;

const DENSITY_OPTIONS = ['comfortable', 'compact'] as const;
const DENSITY_LABELS: Record<GridConfig['information_density'], string> = {
  comfortable: '舒适',
  compact: '紧凑',
};

const MOBILE_MAX_WIDTH = 600;
const MIN_TILE_WIDTH = 200;
const MIN_COLUMNS = 2;
const MAX_COLUMNS = 8;

const FALLBACK_TITLE = '未命名条目';

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function isDensity(value: unknown): value is GridConfig['information_density'] {
  return value === 'compact' || value === 'comfortable';
}

function normalizeConfig(input: unknown): GridConfig {
  const record = isRecord(input) ? input : {};
  const gap = record.gap;
  const showTags = record.show_tags;
  const density = record.information_density;
  return {
    gap:
      typeof gap === 'number' && Number.isFinite(gap)
        ? Math.min(Math.max(Math.round(gap), GAP_MIN), GAP_MAX)
        : DEFAULT_CONFIG.gap,
    show_tags: typeof showTags === 'boolean' ? showTags : DEFAULT_CONFIG.show_tags,
    information_density: isDensity(density) ? density : DEFAULT_CONFIG.information_density,
  };
}

// ===== src/main.ts（主进程纯函数，构建产物 dist/index.js）=====

function validateConfig(config: unknown): boolean {
  if (config === undefined || config === null) {
    return true;
  }
  if (!isRecord(config)) {
    return false;
  }
  const gap = config.gap;
  const showTags = config.show_tags;
  const density = config.information_density;
  if (gap !== undefined && (typeof gap !== 'number' || !Number.isInteger(gap) || gap < GAP_MIN || gap > GAP_MAX)) {
    return false;
  }
  if (showTags !== undefined && typeof showTags !== 'boolean') {
    return false;
  }
  if (density !== undefined && !isDensity(density)) {
    return false;
  }
  return true;
}

// ---------- 响应式列数：由容器宽度自动计算，移动端两列 ----------

function computeColumnCount(containerWidth: number): number {
  if (containerWidth <= MOBILE_MAX_WIDTH) {
    return MIN_COLUMNS;
  }
  const byWidth = Math.floor(containerWidth / MIN_TILE_WIDTH);
  return Math.min(MAX_COLUMNS, Math.max(MIN_COLUMNS, byWidth));
}

// ---------- 条目外壳 ----------

function readTagLabels(snapshot: Record<string, unknown> | undefined): readonly string[] {
  if (snapshot === undefined) {
    return [];
  }
  const tags = snapshot.tags;
  if (!Array.isArray(tags)) {
    return [];
  }
  const labels: string[] = [];
  for (const tag of tags) {
    if (typeof tag === 'string' && tag.length > 0) {
      labels.push(tag);
    } else if (Array.isArray(tag)) {
      const nested = tag.filter((item): item is string => typeof item === 'string' && item.length > 0);
      if (nested.length > 0) {
        labels.push(nested.join(' / '));
      }
    }
  }
  return labels;
}

function describeFailure(error: unknown): string {
  if (isChipsError(error) && error.code === 'BRIDGE_PERMISSION_DENIED') {
    return '缺少打开条目的权限';
  }
  return '条目暂时无法打开，请稍后重试';
}

// ---------- 帧运行时注入与条目帧管理 ----------

let frameRuntime: IBoxLayoutFrameRuntime | null = null;

function attachRuntime(runtime: IBoxLayoutFrameRuntime): void {
  frameRuntime = runtime;
}

function requireRuntime(): IBoxLayoutFrameRuntime {
  if (frameRuntime === null) {
    throw new Error('布局帧运行时尚未注入');
  }
  return frameRuntime;
}

type TileBinding = {
  readonly entry: EntryView;
  readonly slot: HTMLDivElement;
  readonly message: HTMLParagraphElement;
  release: Dispose | null;
  isMounting: boolean;
};

async function activateEntry(binding: TileBinding): Promise<void> {
  try {
    await requireRuntime().openDocument(binding.entry.url);
    binding.message.hidden = true;
  } catch (error: unknown) {
    binding.message.textContent = describeFailure(error);
    binding.message.hidden = false;
  }
}

function createTile(binding: TileBinding, config: GridConfig): HTMLDivElement {
  const tile = document.createElement('article');
  tile.style.display = 'flex';
  tile.style.flexDirection = 'column';
  tile.style.gap = config.information_density === 'compact' ? '4px' : '8px';
  tile.style.minWidth = '0';

  binding.slot.style.minHeight = '80px';
  binding.slot.style.borderRadius = '8px';
  binding.slot.style.overflow = 'hidden';

  const title = document.createElement('button');
  title.type = 'button';
  title.textContent =
    binding.entry.displayTitle.length > 0 ? binding.entry.displayTitle : FALLBACK_TITLE;
  title.style.border = '0';
  title.style.background = 'transparent';
  title.style.padding = '0';
  title.style.cursor = 'pointer';
  title.style.textAlign = 'center';
  title.addEventListener('click', () => {
    void activateEntry(binding);
  });

  tile.append(binding.slot, title);

  if (config.show_tags) {
    const labels = readTagLabels(binding.entry.snapshot);
    if (labels.length > 0) {
      const tagLine = document.createElement('p');
      tagLine.textContent = labels.join(' · ');
      tagLine.style.margin = '0';
      tagLine.style.textAlign = 'center';
      tile.append(tagLine);
    }
  }

  binding.message.hidden = true;
  binding.message.style.margin = '0';
  binding.message.style.textAlign = 'center';
  tile.append(binding.message);
  return tile;
}

function createGridLayout(box: BoxRenderView, container: HTMLElement): Dispose {
  const config = normalizeConfig(box.layout_config);
  const entries = box.entries.filter((entry) => entry.enabled);

  const root = document.createElement('div');
  root.style.minHeight = '100%';
  root.style.padding = config.information_density === 'compact' ? '12px' : '16px';
  root.style.boxSizing = 'border-box';

  if (entries.length === 0) {
    const empty = document.createElement('p');
    empty.textContent = '暂无条目';
    empty.style.textAlign = 'center';
    root.append(empty);
    container.replaceChildren(root);
    return () => {
      root.remove();
    };
  }

  const grid = document.createElement('div');
  grid.style.display = 'grid';
  grid.style.gap = `${config.gap}px`;
  grid.style.alignItems = 'start';

  const bindings = new Map<string, TileBinding>();
  for (const entry of entries) {
    const slot = document.createElement('div');
    slot.dataset.entryId = entry.entry_id;
    const message = document.createElement('p');
    const binding: TileBinding = { entry, slot, message, release: null, isMounting: false };
    bindings.set(entry.entry_id, binding);
    grid.append(createTile(binding, config));
  }
  root.append(grid);
  container.replaceChildren(root);

  let columns = computeColumnCount(container.clientWidth);
  grid.style.gridTemplateColumns = `repeat(${columns}, minmax(0, 1fr))`;

  const releaseEntry = (entryId: string): void => {
    const binding = bindings.get(entryId);
    if (binding === undefined || binding.release === null) {
      return;
    }
    const release = binding.release;
    binding.release = null;
    release();
  };

  const mountEntry = (binding: TileBinding): void => {
    if (binding.release !== null || binding.isMounting || !binding.slot.isConnected) {
      return;
    }
    binding.isMounting = true;
    void requireRuntime()
      .requestEntries({ entryIds: [binding.entry.entry_id] })
      .then(async (tickets) => {
        if (tickets.length === 0 || binding.slot.isConnected === false) {
          return;
        }
        const release = await requireRuntime().mountEntry(tickets[0], binding.slot);
        if (binding.slot.isConnected) {
          binding.release = release;
        } else {
          release();
        }
      })
      .catch(() => {
        binding.message.textContent = '条目内容暂时不可用';
        binding.message.hidden = false;
      })
      .finally(() => {
        binding.isMounting = false;
      });
  };

  const visibilityObserver = new IntersectionObserver((records) => {
    for (const record of records) {
      const slot = record.target as HTMLElement;
      const entryId = slot.dataset.entryId;
      if (entryId === undefined) {
        continue;
      }
      if (record.isIntersecting) {
        const binding = bindings.get(entryId);
        if (binding !== undefined) {
          mountEntry(binding);
        }
      } else {
        releaseEntry(entryId);
      }
    }
  });

  const widthObserver = new ResizeObserver((records) => {
    for (const record of records) {
      const next = computeColumnCount(record.contentRect.width);
      if (next !== columns) {
        columns = next;
        grid.style.gridTemplateColumns = `repeat(${columns}, minmax(0, 1fr))`;
      }
    }
  });
  widthObserver.observe(container);

  for (const binding of bindings.values()) {
    visibilityObserver.observe(binding.slot);
  }

  return () => {
    visibilityObserver.disconnect();
    widthObserver.disconnect();
    for (const binding of bindings.values()) {
      if (binding.release !== null) {
        binding.release();
        binding.release = null;
      }
    }
    bindings.clear();
    root.remove();
  };
}

type NumberField = {
  readonly element: HTMLLabelElement;
  readonly input: HTMLInputElement;
};

function createNumberField(label: string, value: number, min: number, max: number): NumberField {
  const element = document.createElement('label');
  element.style.display = 'flex';
  element.style.flexDirection = 'column';
  element.style.gap = '4px';

  const text = document.createElement('span');
  text.textContent = label;

  const input = document.createElement('input');
  input.type = 'number';
  input.min = String(min);
  input.max = String(max);
  input.value = String(value);

  element.append(text, input);
  return { element, input };
}

type SelectField = {
  readonly element: HTMLLabelElement;
  readonly select: HTMLSelectElement;
};

function createSelectField(
  label: string,
  options: readonly { value: string; label: string }[],
  value: string,
): SelectField {
  const element = document.createElement('label');
  element.style.display = 'flex';
  element.style.flexDirection = 'column';
  element.style.gap = '4px';

  const text = document.createElement('span');
  text.textContent = label;

  const select = document.createElement('select');
  for (const option of options) {
    const item = document.createElement('option');
    item.value = option.value;
    item.textContent = option.label;
    select.append(item);
  }
  select.value = value;

  element.append(text, select);
  return { element, select };
}

function createSwitchField(label: string, checked: boolean): SwitchField {
  const element = document.createElement('label');
  element.style.display = 'flex';
  element.style.alignItems = 'center';

  input.type = 'checkbox';

  return { element, input };

class GridEditor implements ILayoutEditor {
  public state: EditorState = 'active';

  private base: GridConfig;
  private revision: number;
  private pending: LayoutChange | null = null;
  private readonly root: HTMLDivElement;
  private readonly gapField: NumberField;
  private readonly densityField: SelectField;
  private readonly tagsField: SwitchField;
  private readonly changeHandlers = new Set<(payload: ILayoutEditorEventMap['change']) => void>();
  private readonly dirtyHandlers = new Set<(payload: ILayoutEditorEventMap['dirty']) => void>();

  public constructor(box: BoxRenderView, container: HTMLElement) {
    this.base = normalizeConfig(box.layout_config);
    this.revision = box.revision;

    this.root = document.createElement('div');
    this.root.style.display = 'flex';
    this.root.style.flexDirection = 'column';
    this.root.style.gap = '12px';
    this.root.style.width = '100%';
    this.root.style.height = '100%';
    this.root.style.minHeight = '0';
    this.root.style.padding = '16px';
    this.root.style.boxSizing = 'border-box';
    this.root.style.overflow = 'auto';

    this.gapField = createNumberField('间距（px）', this.base.gap, GAP_MIN, GAP_MAX);
    this.densityField = createSelectField(
      '信息密度',
      DENSITY_OPTIONS.map((value) => ({ value, label: DENSITY_LABELS[value] })),
      this.base.information_density,
    );
    this.tagsField = createSwitchField('显示标签', this.base.show_tags);

    this.root.append(this.gapField.element, this.densityField.element, this.tagsField.element);
    container.replaceChildren(this.root);

    this.gapField.input.addEventListener('input', this.handleInput);
    this.densityField.select.addEventListener('change', this.handleInput);
    this.tagsField.input.addEventListener('change', this.handleInput);
  }

  public isDirty(): boolean {
    return this.pending !== null;
  }

  public getChange(): LayoutChange | null {
    return this.pending;
  }

  public markCommitted(revision: number): void {
    if (this.state === 'disposed' || revision <= this.revision) {
      return;
    }
    const committed = this.pending;
    this.revision = revision;
    if (committed !== null && committed.kind === 'replace') {
      this.base = normalizeConfig(committed.config);
    }
    if (this.pending === committed) {
      this.pending = null;
      this.setDirty(false);
    }
  }

  public update(box: BoxRenderView, revision: number): void {
    if (this.state === 'disposed' || revision <= this.revision) {
      return;
    }
    this.revision = revision;
    this.base = normalizeConfig(box.layout_config);
    this.pending = null;
    this.setDirty(false);
    this.renderFromConfig(this.base);
  }

  public subscribe<TEvent extends keyof ILayoutEditorEventMap>(
    event: TEvent,
    handler: (payload: ILayoutEditorEventMap[TEvent]) => void,
  ): ISubscription {
    if (this.state === 'disposed') {
      return { dispose: (): void => undefined };
    }
    if (event === 'change') {
      const typed = handler as (payload: ILayoutEditorEventMap['change']) => void;
      this.changeHandlers.add(typed);
      return {
        dispose: (): void => {
          this.changeHandlers.delete(typed);
        },
      };
    }
    const typed = handler as (payload: ILayoutEditorEventMap['dirty']) => void;
    this.dirtyHandlers.add(typed);
    return {
      dispose: (): void => {
        this.dirtyHandlers.delete(typed);
      },
    };
  }

  public focus(): void {
    if (this.state === 'active') {
      this.gapField.input.focus();
    }
  }

  public dispose(): void {
    if (this.state === 'disposed') {
      return;
    }
    this.state = 'disposed';
    this.gapField.input.removeEventListener('input', this.handleInput);
    this.densityField.select.removeEventListener('change', this.handleInput);
    this.tagsField.input.removeEventListener('change', this.handleInput);
    this.changeHandlers.clear();
    this.dirtyHandlers.clear();
    this.pending = null;
    this.root.remove();
  }

  private readonly handleInput = (): void => {
    if (this.state === 'disposed') {
      return;
    }
    const next = normalizeConfig({
      gap: this.gapField.input.valueAsNumber,
      show_tags: this.tagsField.input.checked,
      information_density: this.densityField.select.value,
    });
    const change: LayoutChange = {
      kind: 'replace',
      config: { ...next },
      schemaVersion: CONFIG_SCHEMA_VERSION,
    };
    const wasDirty = this.pending !== null;
    this.pending = change;
    if (!wasDirty) {
      this.setDirty(true);
    }
    for (const handler of this.changeHandlers) {
      handler({ change });
    }
  };

  private renderFromConfig(config: GridConfig): void {
    this.gapField.input.value = String(config.gap);
    this.densityField.select.value = config.information_density;
    this.tagsField.input.checked = config.show_tags;
  }

  private setDirty(isDirty: boolean): void {
    for (const handler of this.dirtyHandlers) {
      handler({ isDirty });
    }
  }
}

function renderEditor(box: BoxRenderView, container: HTMLElement): ILayoutEditor {
  return new GridEditor(box, container);
}

  attachRuntime,
  renderLayout,
  renderEditor,
};
