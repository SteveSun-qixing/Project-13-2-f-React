### 9.1 源码目录

```text
richtext-basecard/
├── manifest.yaml
├── package.json                     # name: @chips/plugin-richtext-basecard；version 与 manifest 一致
├── tsconfig.json
├── src/
│   ├── config.ts                    # 共享类型与纯函数
│   ├── main.ts                      # 主进程入口源码 → dist/index.js
│   ├── render.tsx                   # 渲染入口源码 → render.js
│   └── index.ts                     # 拆分导出
├── dist/
│   ├── index.js                     # 主进程入口（受管模块加载器加载）
│   └── render.js                    # 渲染入口（渲染帧装载）
└── assets/
    ├── icons/
    │   └── richtext.svg
    └── schemas/
        └── richtext-config.schema.json
```
