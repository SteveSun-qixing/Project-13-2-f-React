# ADR-003: Rust 核心 + Node Native Module 架构

## 职责划分

### Rust 核心

```rust
// Cordis 运行时
pub struct CordisRuntime {
    context: Arc<Context>,
    plugins: HashMap<String, Plugin>,
}

impl CordisRuntime {
    pub fn load_plugin(&mut self, path: &Path) -> Result<()>;
    pub fn unload_plugin(&mut self, id: &str) -> Result<()>;
    pub fn get_service(&self, name: &str) -> Option<&dyn Service>;
}

// Bridge 实现
pub struct Bridge {
    capabilities: HashMap<String, Box<dyn Capability>>,
}

impl Bridge {
    pub async fn request(&self, id: &str) -> Result<CapabilityHandle>;
    pub fn register(&mut self, id: String, cap: Box<dyn Capability>);
}

// 文档解析
pub fn parse_card(path: &Path) -> Result<Card>;
pub fn parse_box(path: &Path) -> Result<Box>;
pub fn validate_metadata(data: &[u8]) -> Result<Metadata>;

// 资源管理
pub struct ResourceManager {
    cache: LruCache<String, Vec<u8>>,
}

impl ResourceManager {
    pub fn load(&mut self, uri: &str) -> Result<Vec<u8>>;
    pub fn cache(&mut self, uri: String, data: Vec<u8>);
    pub fn clear(&mut self);
}
```

### Node.js 层

```typescript
// Electron 应用逻辑
import { app, BrowserWindow, ipcMain } from 'electron';
import * as native from '../native';

// 窗口管理
class WindowManager {
  private windows: Map<string, BrowserWindow>;
  
  createWindow(id: string, options: WindowOptions): BrowserWindow;
  closeWindow(id: string): void;
  getWindow(id: string): BrowserWindow | undefined;
}

// IPC 处理
ipcMain.handle('bridge:request', async (event, capabilityId) => {
  const result = await native.bridgeRequest(capabilityId);
  return result;
});

ipcMain.handle('plugin:load', async (event, pluginPath) => {
  native.loadPlugin(pluginPath);
});
```
