# ADR-004: 能力导向 Bridge 设计

## 参数验证

### JSON Schema 验证

```typescript
// 能力定义
const fileReadCapability = {
  id: 'file:read',
  schema: {
    input: {
      type: 'object',
      properties: {
        path: { type: 'string', minLength: 1 },
        encoding: { 
          type: 'string', 
          enum: ['utf-8', 'ascii', 'base64'],
          default: 'utf-8',
        },
      },
      required: ['path'],
    },
    output: {
      type: 'string',
    },
  },
  execute: async (input: { path: string; encoding?: string }) => {
    return fs.readFileSync(input.path, input.encoding);
  },
};

// Bridge 自动验证
await chips.request('file:read', { path: '' });  // ❌ 错误：path 长度不足
await chips.request('file:read', { path: 'file.txt' });  // ✅ 正确
await chips.request('file:read', {
  path: 'file.txt',
  encoding: 'invalid',
});  // ❌ 错误：encoding 不在枚举中
```
