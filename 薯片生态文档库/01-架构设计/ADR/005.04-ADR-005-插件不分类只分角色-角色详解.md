# ADR-005: 插件不分类只分角色

## 角色详解

### BaseCard Plugin

```yaml
# manifest.yaml
id: chips.basecard.richtext
type: basecard
runtime:
  targets: [host]

capabilities:
  cardTypes:
    - id: base.richtext
      name: 富文本卡片
      icon: assets/icons/richtext.svg
```

**实现接口**：

```typescript
interface IBaseCardPlugin {
  // 渲染查看器
  renderView(card: ICard, container: HTMLElement): void;
  
  // 渲染编辑器
  renderEditor(card: ICard, container: HTMLElement): IEditor;
  
  // 验证配置
  validateConfig(config: unknown): boolean;
  
  // 导出
  export?(card: ICard, format: string): Promise<Blob>;
}
```

---

### Box Layout Plugin

```yaml
# manifest.yaml
id: chips.layout.grid
type: boxlayout
runtime:
  targets: [host]

capabilities:
  layouts:
    - id: chips.layout.grid
      name: 网格布局
      icon: assets/icons/grid.svg
```

**实现接口**：

```typescript
interface IBoxLayoutPlugin {
  // 渲染布局
  renderLayout(box: IBox, container: HTMLElement): void;
  
  // 渲染编辑器
  renderEditor(box: IBox, container: HTMLElement): IEditor;
  
  // 验证配置
  validateConfig(config: unknown): boolean;
}
```

---

### App Plugin

```yaml
# manifest.yaml
id: chips.app.file-manager
type: app
runtime:
  targets: [host]

ui:
  surface: window          # window | desktop | tray
  launcher:
    icon: assets/icons/app-icon.svg
    name: 文件管理器
```

**实现接口**：

```typescript
interface IAppPlugin {
  // 启动应用
  launch(ctx: IAppContext): Promise<void>;
  
  // 处理命令
  handleCommand?(command: string, args: unknown): Promise<void>;
  
  // 处理文件打开
  openFile?(path: string): Promise<void>;
}
```

---

### Service Plugin

```yaml
# manifest.yaml
id: chips.service.sync
type: service
runtime:
  targets: [host]
  background: true
```

**实现接口**：

```typescript
interface IServicePlugin {
  // 启动服务
  start(ctx: IServiceContext): Promise<void>;
  
  // 停止服务
  stop(): Promise<void>;
  
  // 健康检查
  health?(): Promise<boolean>;
}
```
