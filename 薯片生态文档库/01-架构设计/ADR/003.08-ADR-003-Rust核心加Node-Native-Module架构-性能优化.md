### 批量操作

```rust
// 批量解析（减少跨语言调用）
fn parse_multiple_cards(mut cx: FunctionContext) -> JsResult<JsArray> {
    let paths = cx.argument::<JsArray>(0)?;
    let len = paths.len(&mut cx);
    
    let result = cx.empty_array();
    
    for i in 0..len {
        let path: Handle<JsString> = paths.get(&mut cx, i)?;
        let card = parse_card(Path::new(&path.value(&mut cx)))
            .or_else(|e| cx.throw_error(e.to_string()))?;
        
        let obj = card_to_js_object(&mut cx, card)?;
        result.set(&mut cx, i, obj)?;
    }
    
    Ok(result)
}
```

### 零拷贝

```rust
// 使用 Buffer 零拷贝传输
fn read_file_zero_copy(mut cx: FunctionContext) -> JsResult<JsBuffer> {
    let path = cx.argument::<JsString>(0)?.value(&mut cx);
    let data = std::fs::read(path)
        .or_else(|e| cx.throw_error(e.to_string()))?;
    
    // 直接传递所有权，零拷贝
    let mut buffer = cx.buffer(data.len())?;
    cx.borrow_mut(&mut buffer, |buf| {
        buf.as_mut_slice().copy_from_slice(&data);
    });
    
    Ok(buffer)
}
```
