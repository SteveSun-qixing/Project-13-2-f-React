# ADR-002: Electron 作为桌面端壳

## 进程模型

### 主进程

```typescript
// main.ts
import { app, BrowserWindow } from 'electron';
import { initCordis } from './cordis';
import { initBridge } from './bridge';

app.on('ready', async () => {
  // 初始化 Cordis（Rust 实现）
  await initCordis();
  
  // 初始化 Bridge
  await initBridge();
  
  // 创建窗口
  const win = new BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,
      sandbox: true,
      preload: path.join(__dirname, 'preload.js'),
    },
  });
  
  win.loadFile('index.html');
});
```

### 预加载脚本

```typescript
// preload.ts
import { contextBridge, ipcRenderer } from 'electron';
import type { IResourceHandle, ISubscription } from '@chips/sdk';
import { createResourceHandle } from './bridge/resource-handle';

// 暴露 Bridge API：window.chips 与 IChipsBridge 同形状（R2）
contextBridge.exposeInMainWorld('chips', {
  // 能力请求：输入随请求携带，返回能力执行结果
  request: (capabilityId: string, input?: unknown): Promise<unknown> =>
    ipcRenderer.invoke('bridge:request', capabilityId, input),
  
  // 事件订阅：返回可退订句柄
  subscribe: (eventId: string, handler: (payload: unknown) => void): ISubscription => {
    const listener = (_event: unknown, payload: unknown): void => handler(payload);
    ipcRenderer.on(`event:${eventId}`, listener);
    return {
      dispose: (): void => {
        ipcRenderer.removeListener(`event:${eventId}`, listener);
      },
    };
  },
  
  // 资源访问：返回资源句柄
  resource: (uri: string): IResourceHandle => createResourceHandle(uri),
});
```

### 渲染进程

```typescript
// renderer/index.tsx
import React from 'react';
import { createRoot } from 'react-dom/client';
import { App } from './App';

// 使用 window.chips（由 preload.ts 注入）
const bridge = window.chips;

const root = createRoot(document.getElementById('root')!);
root.render(<App bridge={bridge} />);
```
