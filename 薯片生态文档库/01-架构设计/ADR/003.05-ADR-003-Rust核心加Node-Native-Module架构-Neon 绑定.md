# ADR-003: Rust 核心 + Node Native Module 架构

## Neon 绑定

### 基础类型转换

```rust
use neon::prelude::*;

// Rust String → JS String
fn rust_to_js_string(cx: &mut FunctionContext, s: String) -> JsString {
    cx.string(s)
}

// JS String → Rust String
fn js_to_rust_string(cx: &mut FunctionContext, js: Handle<JsString>) -> String {
    js.value(cx)
}

// Rust Vec<u8> → JS Buffer
fn rust_to_js_buffer<'a>(cx: &mut FunctionContext<'a>, data: Vec<u8>) -> JsResult<'a, JsBuffer> {
    let mut buffer = cx.buffer(data.len())?;
    cx.borrow_mut(&mut buffer, |buf| {
        buf.as_mut_slice().copy_from_slice(&data);
    });
    Ok(buffer)
}

// JS Object → Rust Struct
fn js_to_rust_config(cx: &mut FunctionContext, obj: Handle<JsObject>) -> Config {
    let enable_cache: Handle<JsBoolean> = obj.get(cx, "enableCache")?;
    let timeout: Handle<JsNumber> = obj.get(cx, "timeout")?;
    
    Config {
        enable_cache: enable_cache.value(cx),
        timeout: timeout.value(cx) as u64,
    }
}
```

### 异步函数

```rust
use neon::prelude::*;
use neon::event::Channel;

// 异步加载卡片
fn load_card_async(mut cx: FunctionContext) -> JsResult<JsPromise> {
    let path = cx.argument::<JsString>(0)?.value(&mut cx);
    let channel = cx.channel();
    
    let (deferred, promise) = cx.promise();
    
    // 在后台线程执行
    std::thread::spawn(move || {
        let result = parse_card(Path::new(&path));
        
        // 返回主线程
        deferred.settle_with(&channel, move |mut cx| {
            match result {
                Ok(card) => {
                    let obj = card_to_js_object(&mut cx, card)?;
                    Ok(obj)
                }
                Err(e) => cx.throw_error(e.to_string()),
            }
        });
    });
    
    Ok(promise)
}

#[neon::main]
fn main(mut cx: ModuleContext) -> NeonResult<()> {
    cx.export_function("loadCardAsync", load_card_async)?;
    Ok(())
}
```

```rust
// Rust 类
pub struct CardParser {
    options: ParseOptions,
}

impl CardParser {
    fn new(options: ParseOptions) -> Self {
        Self { options }
    }
    
    fn parse(&self, path: &Path) -> Result<Card> {
        // 实现
    }
}

// JS 类
impl Finalize for CardParser {}

impl CardParser {
    fn js_new(mut cx: FunctionContext) -> JsResult<JsBox<CardParser>> {
        let options_obj = cx.argument::<JsObject>(0)?;
        let options = js_to_parse_options(&mut cx, options_obj)?;
        
        let parser = CardParser::new(options);
        Ok(cx.boxed(parser))
    }
    
    fn js_parse(mut cx: FunctionContext) -> JsResult<JsObject> {
        let parser = cx.argument::<JsBox<CardParser>>(0)?;
        let path = cx.argument::<JsString>(1)?.value(&mut cx);
        
        let card = parser.parse(Path::new(&path))
            .or_else(|e| cx.throw_error(e.to_string()))?;
        
        card_to_js_object(&mut cx, card)
    }
}

#[neon::main]
fn main(mut cx: ModuleContext) -> NeonResult<()> {
    cx.export_function("CardParser", CardParser::js_new)?;
    cx.export_function("cardParserParse", CardParser::js_parse)?;
    Ok(())
}
```
