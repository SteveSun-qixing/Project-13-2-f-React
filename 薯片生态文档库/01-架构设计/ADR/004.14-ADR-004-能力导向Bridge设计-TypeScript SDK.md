# ADR-004: 能力导向 Bridge 设计

## TypeScript SDK

### chips-sdk 封装

```typescript
// @chips/sdk
export class ChipsBridge implements IChipsBridge {
  request<TOutput>(capabilityId: string, input?: unknown): Promise<TOutput> {
    // 通过 window.chips 调用
    return window.chips.request<TOutput>(capabilityId, input);
  }
  
  subscribe(eventId: string, handler: (payload: unknown) => void): ISubscription {
    return window.chips.subscribe(eventId, handler);
  }
  
  resource(uri: string): IResourceHandle {
    return window.chips.resource(uri);
  }
}

// 插件使用
import { chips } from '@chips/sdk';

const data = await chips.request<string>('file:read', { path: 'file.txt' });
```
