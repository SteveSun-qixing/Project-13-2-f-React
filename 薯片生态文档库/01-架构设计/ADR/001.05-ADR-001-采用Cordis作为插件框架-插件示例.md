# ADR-001: 采用 Cordis 作为插件框架

## 插件示例

### 基础插件

```typescript
export class RichTextCardPlugin {
  readonly id = 'chips.basecard.richtext';
  
  constructor(ctx: Context) {
    // 注册富文本卡片服务（服务名不用卡片类型 ID）
    ctx.provide('richtext-card', this);
    
    // 依赖文件服务
    ctx.inject(['file'], (ctx) => {
      ctx.on('card:render', async (card) => {
        if (card.type === 'base.richtext') {
          return this.render(card);
        }
      });
    });
  }
  
  render(card: Card): Html {
    // 渲染逻辑
  }
}
```

### 热插拔

```typescript
// 加载插件
const fork = ctx.plugin(RichTextCardPlugin);

// 运行中...

// 卸载插件
fork.dispose();

// 重新加载
const newFork = ctx.plugin(RichTextCardPlugin);
```
