# ADR-004: 能力导向 Bridge 设计

## 类型系统

### 能力类型定义

```typescript
// file:read 能力
interface IFileReadCapability extends ICapability<
  { path: string; encoding?: string },
  string
> {
  id: 'file:read';
}

// dialog:save 能力
interface IDialogSaveCapability extends ICapability<
  { defaultPath?: string; filters?: IFileFilter[] },
  { path: string; canceled: boolean }
> {
  id: 'dialog:save';
}

// window:create 能力
interface IWindowCreateCapability extends ICapability<
  { width: number; height: number; title: string },
  IWindow
> {
  id: 'window:create';
}
```

### 使用示例

```typescript
// 插件代码（类型安全：以执行结果为泛型参数）
async function readAndParse(configPath: string): Promise<IConfig> {
  // 请求能力（返回类型明确）
  const data = await chips.request<string>('file:read', { path: configPath });
  const config = JSON.parse(data) as IConfig;
  
  return config;
}
```
