### 示例：读取卡片标题

​```typescript
import { chips } from '@chips/sdk';
import type { CapabilityOutput } from '@chips/sdk';

// 打开文档后读取 metadata（路径为包内相对路径或受管 URI）
const opened = await chips.request<CapabilityOutput<'document:open'>>('document:open', {
  source: 'chips-resource://card-7f3a',
});
const metadata = await chips.request<CapabilityOutput<'document:read'>>('document:read', {
  documentId: opened.documentId,
  part: 'metadata',
});

console.log(metadata.value);   // 卡片元数据切片
console.log(opened.title);     // 卡片标题
​```
```
