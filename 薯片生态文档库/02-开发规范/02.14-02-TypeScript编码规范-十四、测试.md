## 十四、测试

### 14.1 单元测试

```typescript
// card.test.ts（与被测文件 card.ts 同目录）
import { describe, it, expect } from 'vitest';
import { parseCard } from './card';

describe('parseCard', () => {
  it('should parse valid card', () => {
    const input = '{"id": "test", "title": "Test"}';
    const card = parseCard(input);
    expect(card.id).toBe('test');
    expect(card.title).toBe('Test');
  });
  
  it('should throw on invalid input', () => {
    expect(() => parseCard('invalid')).toThrow(ChipsError);
  });
});
```
