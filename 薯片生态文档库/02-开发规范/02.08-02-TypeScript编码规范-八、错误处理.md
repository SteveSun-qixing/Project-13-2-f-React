## 八、错误处理

### 8.1 统一错误类

全生态只有一个错误类 `ChipsError`（由 `@chips/sdk` 提供），禁止按错误码派生子类：

```typescript
interface IChipsErrorOptions {
  code: string;        // <DOMAIN>_<REASON>，如 DOCUMENT_NOT_FOUND
  message: string;     // 简体中文、面向用户
  details?: unknown;   // 结构化诊断
  retryable?: boolean; // 默认 false
}

class ChipsError extends Error {
  public readonly code: string;
  public readonly details?: unknown;
  public readonly retryable: boolean;

  constructor(options: IChipsErrorOptions) {
    super(options.message);
    this.name = 'ChipsError';
    this.code = options.code;
    this.details = options.details;
    this.retryable = options.retryable ?? false;
  }
}

// 抛出与识别：用错误码区分原因，不派生子类
throw new ChipsError({
  code: 'DOCUMENT_NOT_FOUND',
  message: `未找到卡片：${cardId}`,
  details: { cardId },
  retryable: false,
});
```

### 8.2 Result 类型

```typescript
// Result 类型
type Result<T, E = Error> = 
  | { success: true; value: T }
  | { success: false; error: E };

// 使用 Result
function parseCard(input: string): Result<Card, ParseError> {
  try {
    const card = JSON.parse(input);
    return { success: true, value: card };
  } catch (error) {
    return { 
      success: false, 
      error: new ParseError('Invalid JSON', error) 
    };
  }
}

// 处理 Result
const result = parseCard(input);
if (result.success) {
  console.log(result.value.title);
} else {
  console.error(result.error.message);
}
```
