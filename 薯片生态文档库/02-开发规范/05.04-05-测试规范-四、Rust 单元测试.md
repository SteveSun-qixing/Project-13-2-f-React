## 四、Rust 单元测试

### 4.1 基本测试

```rust
#[cfg(test)]
mod tests {
    use super::*;
    
    #[test]
    fn test_parse_card() {
        let input = r#"{"id": "test", "title": "Test"}"#;
        let card = parse_card(input).unwrap();
        
        assert_eq!(card.id, "test");
        assert_eq!(card.title, "Test");
    }
    
    #[test]
    #[should_panic(expected = "Invalid JSON")]
    fn test_invalid_input() {
        parse_card("invalid").unwrap();
    }
    
    #[test]
    fn test_error_handling() {
        let result = parse_card("invalid");
        assert!(result.is_err());
        
        match result {
            Err(CardError::InvalidFormat(_)) => {},
            _ => panic!("Expected InvalidFormat error"),
        }
    }
}
```

### 4.2 异步测试

```rust
#[cfg(test)]
mod tests {
    use super::*;
    
    #[tokio::test]
    async fn test_load_card() {
        let card = load_card("test.card").await.unwrap();
        assert_eq!(card.id, "test");
    }
    
    #[tokio::test]
    async fn test_concurrent_load() {
        let (card1, card2) = tokio::join!(
            load_card("test1.card"),
            load_card("test2.card")
        );
        
        assert!(card1.is_ok());
        assert!(card2.is_ok());
    }
}
```

### 4.3 属性测试

```rust
use proptest::prelude::*;

proptest! {
    #[test]
    fn test_parse_roundtrip(id in "\\w+", title in ".*") {
        let card = Card { id, title };
        let serialized = serialize_card(&card).unwrap();
        let deserialized = deserialize_card(&serialized).unwrap();
        
        prop_assert_eq!(card, deserialized);
    }
}
```
