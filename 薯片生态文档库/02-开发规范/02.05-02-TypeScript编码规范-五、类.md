```typescript
class CardParser {
  // 私有字段
  private readonly options: IParseOptions;
  
  // 公开字段
  public readonly version: string;
  
  // 构造函数
  constructor(options: IParseOptions) {
    this.options = options;
    this.version = '1.0.0';
  }
  
  // 公开方法
  public parse(input: string): Card {
    return this.parseInternal(input);
  }
  
  // 私有方法
  private parseInternal(input: string): Card {
    // ...
  }
  
  // 静态方法
  public static create(options: IParseOptions): CardParser {
    return new CardParser(options);
  }
}
```
