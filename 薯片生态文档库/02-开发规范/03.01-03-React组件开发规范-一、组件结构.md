## 一、组件结构

### 1.1 文件组织

```
ComponentName/
├── index.tsx              # 组件主文件
├── types.ts               # 类型定义
├── hooks.ts               # 自定义 Hooks
├── utils.ts               # 工具函数
└── index.test.tsx         # 测试文件（与被测文件同目录）
```

组件目录内不得出现 `styles.css`、`*.module.css` 等样式文件：样式唯一来源是 `@chips/ui` 组件与 Design Tokens（第十一章）。

### 1.2 组件模板

```tsx
// ComponentName/index.tsx
import React from 'react';
import { theme } from '@chips/ui';
import type { ComponentNameProps } from './types';

export const ComponentName: React.FC<ComponentNameProps> = ({
  prop1,
  prop2,
  ...rest
}) => {
  // Hooks
  const [state, setState] = React.useState<Type>(initialValue);

  // Design Tokens（视觉数值唯一样式来源）
  const { token } = theme.useToken();

  // Effects
  React.useEffect(() => {
    // Effect logic
  }, [dependencies]);

  // Handlers
  const handleEvent = React.useCallback(() => {
    // Handler logic
  }, [dependencies]);

  // Render
  return (
    <section {...rest} style={{ padding: token.padding, color: token.colorText }}>
      {/* Content */}
    </section>
  );
};
```
