## 三、Hooks 使用

### 3.1 Hooks 顺序

```tsx
export const Component: React.FC<Props> = (props) => {
  // 1. useState
  const [count, setCount] = useState(0);
  const [text, setText] = useState('');

  // 2. useRef
  const ref = useRef<HTMLDivElement>(null);

  // 3. Design Tokens（antd 主题经 @chips/ui 暴露）
  const { token } = theme.useToken();

  // 4. useMemo
  const computed = useMemo(() => {
    return expensiveComputation(count);
  }, [count]);

  // 5. useCallback
  const handleClick = useCallback(() => {
    setCount((current) => current + 1);
  }, []);

  // 6. useEffect
  useEffect(() => {
    // Side effect
  }, [dependencies]);

  // 7. 自定义 Hooks
  const data = useCustomHook(id);

  // Render
  return <div style={{ color: token.colorPrimary }}>...</div>;
};
```

### 3.2 自定义 Hooks

```typescript
// hooks.ts
export function useCardData(cardId: string) {
  const [data, setData] = useState<Card | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);
  
  useEffect(() => {
    let cancelled = false;
    
    setLoading(true);
    loadCard(cardId)
      .then(card => {
        if (!cancelled) {
          setData(card);
          setLoading(false);
        }
      })
      .catch(err => {
        if (!cancelled) {
          setError(err);
          setLoading(false);
        }
      });
    
    return () => {
      cancelled = true;
    };
  }, [cardId]);
  
  return { data, loading, error };
}

// 使用
const { data, loading, error } = useCardData(cardId);
```
