## 七、异步处理

### 7.1 Promise

```typescript
// 返回 Promise
function loadCard(path: string): Promise<Card> {
  return new Promise((resolve, reject) => {
    fs.readFile(path, (err, data) => {
      if (err) reject(err);
      else resolve(parseCard(data));
    });
  });
}

// Promise 链
loadCard('test.card')
  .then((card) => validateCard(card))
  .then((isValid) => console.log(isValid))
  .catch((error) => console.error(error));
```

### 7.2 async/await

```typescript
// async 函数
async function loadAndValidate(path: string): Promise<boolean> {
  try {
    const card = await loadCard(path);
    const isValid = await validateCard(card);
    return isValid;
  } catch (error) {
    console.error('Failed to load card:', error);
    return false;
  }
}

// 并行执行
async function loadMultiple(paths: string[]): Promise<Card[]> {
  const promises = paths.map((path) => loadCard(path));
  return Promise.all(promises);
}
```
