## 二、Props 设计

### 2.1 Props 接口

```typescript
// types.ts
export interface ComponentNameProps {
  // 必需属性
  id: string;
  title: string;
  
  // 可选属性
  description?: string;
  disabled?: boolean;
  
  // 回调函数
  onClick?: (event: React.MouseEvent) => void;
  onChange?: (value: string) => void;
  
  // 子元素
  children?: React.ReactNode;

  // 样式透传：仅用于布局组合；自定义视觉值必须取 Design Tokens
  className?: string;
  style?: React.CSSProperties;
}
```

### 2.2 默认值

```typescript
// 使用解构默认值
export const Component: React.FC<Props> = ({
  disabled = false,
  size = 'medium',
  ...rest
}) => {
  // ...
};

// 或使用 defaultProps（旧方式）
Component.defaultProps = {
  disabled: false,
  size: 'medium',
};
```
