## 二、命名规范

### 2.1 文件命名

```
PascalCase：组件文件、类文件
  CardTile.tsx
  PluginManager.ts
  BridgeClient.ts

kebab-case：工具函数、Hook、配置文件
  use-card-data.ts
  file-utils.ts
  parse-card.ts
```

### 2.2 标识符命名

```typescript
// 接口：PascalCase，I 前缀
interface ICardMetadata {
  id: string;
  title: string;
}

// 类型别名：PascalCase
type CardId = string;
type ParseResult = Card | null;

// 枚举：PascalCase
enum CardType {
  Text = 'text',
  Image = 'image',
  Video = 'video',
}

// 常量：SCREAMING_SNAKE_CASE
const MAX_RETRY_COUNT = 3;
const DEFAULT_TIMEOUT = 30000;

// 函数：camelCase
function parseCard(path: string): Card { }
function validateMetadata(meta: ICardMetadata): boolean { }

// 变量：camelCase
const cardList: Card[] = [];
const isValid = true;

// 类：PascalCase
class CardParser { }
class PluginManager { }

// React 组件：PascalCase
function CardTile(props: CardTileProps) { }
const BoxContainer: React.FC<BoxContainerProps> = ({ children }) => { };
```

### 2.3 布尔命名

```typescript
// 使用 is/has/can/should 前缀
const isLoading = true;
const hasPermission = false;
const canEdit = true;
const shouldRetry = false;

// 禁止的命名
const loading = true;        // ❌
const permission = false;    // ❌
const editable = true;       // ❌
```
