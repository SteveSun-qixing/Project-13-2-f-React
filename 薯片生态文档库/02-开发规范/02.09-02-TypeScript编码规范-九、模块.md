## 九、模块

### 9.1 导入导出

```typescript
// 命名导出
export const VERSION = '1.0.0';
export function parseCard(input: string): Card { }
export class CardParser { }

// 默认导出
export default class Card { }

// 导入
import Card from './Card';
import { parseCard, VERSION } from './parser';
import * as utils from './utils';

// 类型导入
import type { ICardMetadata } from './types';
```

### 9.2 路径别名

```typescript
// 使用路径别名（tsconfig.json 配置）；子路径只在包 exports 声明时使用
import { Button, theme } from '@chips/ui';
import type { ICapability } from '@chips/bridge';

// 禁止相对路径超过两级
import { utils } from './utils';           // ✅

import { parse } from '../../../parser';   // ❌ 使用别名
```
