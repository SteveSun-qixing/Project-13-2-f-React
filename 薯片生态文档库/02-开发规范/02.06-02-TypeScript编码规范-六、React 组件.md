import React from 'react';
import { Card, Spin } from '@chips/ui';

// Props 接口
interface CardTileProps {
  cardId: string;
  preview?: boolean;
  onClick?: () => void;
}

// 函数组件：样式一律来自 @chips/ui 与 Design Tokens，禁止自建 CSS 类
export const CardTile: React.FC<CardTileProps> = ({
  cardId,
  preview = false,
  onClick,
}) => {
  const [loading, setLoading] = React.useState(false);

  React.useEffect(() => {
    void loadCard(cardId);
  }, [cardId]);

  return (
    <Card hoverable onClick={onClick}>
      {loading ? <Spin /> : <CardContent preview={preview} />}
    </Card>
  );
};

### 6.2 Hooks

```typescript
// 自定义 Hook
function useCardData(cardId: string) {
  const [data, setData] = useState<Card | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);
  
  useEffect(() => {
    let cancelled = false;
    
    loadCard(cardId)
      .then((card) => {
        if (!cancelled) {
          setData(card);
          setLoading(false);
        }
      })
      .catch((err) => {
        if (!cancelled) {
          setError(err);
          setLoading(false);
        }
      });
    
    return () => {
      cancelled = true;
    };
  }, [cardId]);
  
  return { data, loading, error };
}

// 使用 Hook
function MyComponent({ cardId }: { cardId: string }) {
  const { data, loading, error } = useCardData(cardId);
  
  if (loading) return <Spin />;
  if (error) return <Error message={error.message} />;
  if (!data) return null;
  
  return <div>{data.title}</div>;
}
```

### 6.3 事件处理

```typescript
interface ButtonProps {
  onClick: (event: React.MouseEvent<HTMLButtonElement>) => void;
  onSubmit?: (data: FormData) => void;
}

const Button: React.FC<ButtonProps> = ({ onClick, onSubmit }) => {
  const handleClick = (event: React.MouseEvent<HTMLButtonElement>) => {
    event.preventDefault();
    onClick(event);
  };
  
  return <button onClick={handleClick}>Click</button>;
};
```
