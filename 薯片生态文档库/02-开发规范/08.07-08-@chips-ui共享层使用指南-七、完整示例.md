## 七、完整示例

示例为一个最小应用插件 `chips.example.doc-viewer`（卡片信息查看器）的窗口 UI：入口挂载 `ChipsProvider` 与 `CommandProvider`，使用 `AppChrome`、`Toolbar`、`StateBoundary`、`StartHome`，注册 ui 目标命令“打开卡片”并挂接本地处理器，经 `chips` 读取自身配置、打开 `.card` 文档并展示信息。主逻辑入口、文件关联与 main 目标命令的完整写法见 [03-应用插件开发指南](../03-插件开发/03-应用插件开发指南.md)；命令系统的完整契约见 [13-命令系统契约](../05-接口与契约/13-命令系统契约.md)；卡片内容的渲染会话挂载见 [05-渲染引擎设计](../01-架构设计/01-子系统设计/05-渲染引擎设计.md) §3.2。

### 7.1 manifest.yaml（片段）

```yaml
id: chips.example.doc-viewer
version: 1.0.0
type: app
name: 卡片信息查看器
description: 打开 .card 文档并展示基本信息
author: 薯片示例
license: MIT
runtime:
  targets: [host]
  background: false
permissions:
  - dialogs:use        # dialog:open 选择文档
  - resources:read     # document:open 读取文档
  - command:write      # 注册 ui 目标命令
  - command:execute    # 调用命令
fileAssociations:
  - extensions: [.card]
    action: open
ui:
  surface: window
  launcher:
    icon: assets/icons/app.svg
    name: 卡片信息查看器
    description: 打开 .card 文档并展示基本信息
```

### 7.2 src/ui/AppRoot.tsx（构建为 render.js）

```tsx
import React from 'react';
import { chips, isChipsError } from '@chips/sdk';
import type { CapabilityInput, CapabilityOutput, ISubscription } from '@chips/sdk';
import {
  AppChrome,
  ChipsProvider,
  CommandProvider,
  StartHome,
  StateBoundary,
  Toolbar,
  formatAccelerator,
  theme,
  toChipsUiFailure,
  useChips,
  useCommandRuntime,
} from '@chips/ui';
import type { ChipsUiFailure } from '@chips/ui';

const APP_ID = 'chips.example.doc-viewer';
const OPEN_COMMAND_ID = 'doc-viewer:open';
const OPEN_HANDLER_ID = 'open-document';
const SHOW_HINT_KEY = 'show-hint';
const CARD_FILTERS = [{ name: '薯片卡片', extensions: ['.card'] }];

const Workspace: React.FC = () => {
  const { token } = theme.useToken();
  const { appId } = useChips();
  const runtime = useCommandRuntime();
  const namespace = `plugins.${appId}`;

  const [document, setDocument] = React.useState<string | null>(null);
  const [opening, setOpening] = React.useState(false);
  const [failure, setFailure] = React.useState<ChipsUiFailure | null>(null);
  const [showHint, setShowHint] = React.useState(true);

  React.useEffect(() => {
    let active = true;

    const load = async (): Promise<void> => {
      try {
        const input: CapabilityInput<'config:get'> = { namespace, key: SHOW_HINT_KEY };
        const result: CapabilityOutput<'config:get'> = await chips.request<
          CapabilityOutput<'config:get'>
        >('config:get', input);
        if (active && typeof result.value === 'boolean') {
          setShowHint(result.value);
        }
      } catch {
        // 读取失败保留默认值，不阻断界面
      }
    };

    void load();

    const subscription: ISubscription = chips.subscribe('config:changed', (payload) => {
      const change = payload as { key?: unknown; effectiveValue?: unknown };
      if (change.key !== `${namespace}.${SHOW_HINT_KEY}`) {
        return;
      }
      if (typeof change.effectiveValue === 'boolean') {
        setShowHint(change.effectiveValue);
      }
    });

    return () => {
      active = false;
      subscription.dispose();
    };
  }, [namespace]);

  const openDocument = React.useCallback((): void => {
    const openCard = async (): Promise<void> => {
      setOpening(true);
      setFailure(null);
      try {
        const dialogInput: CapabilityInput<'dialog:open'> = {
          title: '打开卡片',
          filters: CARD_FILTERS,
        };
        const picked: CapabilityOutput<'dialog:open'> = await chips.request<
          CapabilityOutput<'dialog:open'>
        >('dialog:open', dialogInput);
        if (picked.canceled) {
          return;
        }
        const source = picked.paths[0];
        if (source === undefined) {
          return;
        }
        const openInput: CapabilityInput<'document:open'> = { source, kind: 'card' };
        const opened: CapabilityOutput<'document:open'> = await chips.request<
          CapabilityOutput<'document:open'>
        >('document:open', openInput);
        setDocument(opened.title);
      } catch (error) {
        setFailure(toChipsUiFailure(error, '打开卡片失败，请重试'));
      } finally {
        setOpening(false);
      }
    };

    void openCard();
  }, []);

  React.useEffect(() => {
    void chips.request('command:register', {
      definition: {
        id: OPEN_COMMAND_ID,
        title: '打开卡片',
        icon: 'assets/icons/app.svg',
        shortcut: 'CommandOrControl+O',
        scope: { kind: 'app' },
        handlerId: OPEN_HANDLER_ID,
        group: '文件',
        keywords: ['打开', '卡片'],
        toolbars: [{ toolbarId: 'doc-viewer-main', group: '文件' }],
      },
    });

    const attachment = runtime.attach([
      { commandId: OPEN_COMMAND_ID, handlerId: OPEN_HANDLER_ID, handler: () => openDocument() },
    ]);

    return () => {
      attachment.dispose();
      void chips.request('command:unregister', { commandId: OPEN_COMMAND_ID });
    };
  }, [openDocument, runtime]);

  return (
    <AppChrome
      footer={showHint ? (document === null ? '未打开卡片' : `已打开：${document}`) : undefined}
    >
      <Toolbar toolbarId="doc-viewer-main" />
      <StateBoundary loading={opening} error={failure} onRetry={openDocument}>
        {document === null ? (
          <StartHome
            title="卡片信息查看器"
            description="打开 .card 文档，查看卡片的基本信息"
            formatsText="支持 .card"
            openLabel="打开卡片"
            shortcutText={formatAccelerator('CommandOrControl+O')}
            dragActive={false}
            dragActiveText="松开以打开卡片"
            onOpen={openDocument}
          />
        ) : (
          <section style={{ padding: token.padding, color: token.colorText }}>
            <h2 style={{ margin: 0, fontSize: token.fontSizeHeading5 }}>{document}</h2>
            <p style={{ margin: `${token.marginXS}px 0 0`, color: token.colorTextSecondary }}>
              文档已打开，内容渲染由渲染会话负责。
            </p>
          </section>
        )}
      </StateBoundary>
    </AppChrome>
  );
};

export const AppRoot: React.FC = () => (
  <ChipsProvider appId={APP_ID} surface="window">
    <CommandProvider>
      <Workspace />
    </CommandProvider>
  </ChipsProvider>
);

export default AppRoot;
```

### 7.3 示例要点

- 入口 `AppRoot` 挂载 `ChipsProvider` 与 `CommandProvider` 各一次；业务界面只消费 `useChips()`、`useCommandRuntime()` 与 `theme.useToken()`。
- 命令“打开卡片”为 ui 目标：`command:register` 提交记录（含 `handlerId`）与本地处理器经 `runtime.attach()` 配对；窗口内快捷键 `CommandOrControl+O` 来自命令记录，不用 `shortcut:register`（§3.4）。
- `Toolbar` 按 `toolbarId` 渲染该命令的按钮；`AppChrome` 内置命令面板与面板快捷键；两者触发都经 `command:invoke`。
- 平台访问只经 `chips`（`@chips/sdk`）；`dialog:open`、`document:open`、`command:*` 的权限在 manifest 按最小范围声明；配置读写自身命名空间免权限。
- 订阅在 effect 清理函数中 `dispose()`；命令挂接与注册在清理中成对撤销；样式只使用 Design Tokens 与内联布局，无全局样式与第二套组件库。
- 失败反馈使用 `StateBoundary` 的错误态与简体中文提示；`ChipsError` 经 `toChipsUiFailure` 归一为 `ChipsUiFailure`。
