## 三、类型定义

### 3.1 接口 vs 类型别名

```typescript
// 接口：对象结构
interface ICardMetadata {
  id: string;
  title: string;
  created: Date;
}

// 类型别名：联合类型、交叉类型、基本类型别名
type CardId = string;
type Result = Success | Failure;
type ReadonlyCard = Readonly<Card>;
```

### 3.2 类型注解

```typescript
// 函数参数和返回值必须显式类型
function parseCard(path: string): Card {
  return new Card(path);
}

// 变量类型可以推断时省略
const count = 0;                    // ✅ 推断为 number
const name = 'test';                // ✅ 推断为 string
const list = [1, 2, 3];             // ✅ 推断为 number[]

// 无法推断时必须注解
const data: ICardMetadata = JSON.parse(str);
const map: Map<string, Card> = new Map();
```

### 3.3 泛型

```typescript
// 泛型函数
function identity<T>(value: T): T {
  return value;
}

// 泛型接口
interface IResponse<T> {
  data: T;
  error: string | null;
}

// 泛型约束
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}

// 泛型默认值
interface ICache<T = unknown> {
  get(key: string): T | undefined;
  set(key: string, value: T): void;
}
```

### 3.4 联合类型与交叉类型

```typescript
// 联合类型
type Status = 'pending' | 'success' | 'error';
type Result = Card | Error;

// 交叉类型
type CardWithMeta = Card & { metadata: ICardMetadata };

// 类型守卫
function isCard(value: Card | Error): value is Card {
  return (value as Card).id !== undefined;
}

if (isCard(result)) {
  console.log(result.id);  // TypeScript 知道这是 Card
}
```
