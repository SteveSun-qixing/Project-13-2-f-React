```
Chips-Notes/

│   ├── services/                       # 业务服务层
│   ├── context/                        # 状态管理层
│   ├── basecards/                      # 基础卡片系统
│   │   ├── types.ts                    # 数据定义
│   │   ├── registry.tsx                # 注册表 + 图标
│   │   ├── renderers.tsx               # 8 种渲染器

│   ├── styles/                         # 全局样式
│   └── runtime/                        # SDK 运行时
└── tests/unit/
    ├── notebook-service.test.ts        # 笔记本测试 (9)
    ├── note-service.test.ts            # 笔记测试 (15)
    └── basecards-registry.test.ts      # 卡片注册测试 (11)
```
