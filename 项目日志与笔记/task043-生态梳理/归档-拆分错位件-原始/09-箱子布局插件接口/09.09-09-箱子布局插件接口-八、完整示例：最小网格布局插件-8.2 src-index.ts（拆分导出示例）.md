### 8.2 `src/index.ts`（拆分导出示例）

下列实现按双入口拆分：`validateConfig` 进入 `dist/index.js`（主进程纯函数），`renderLayout` / `renderEditor` / `attachRuntime` 进入 `render.js`（渲染入口）；源码入口只做拆分导出，构建按两入口分别打包。

```typescript
import type {
  BoxRenderView,
  Dispose,
  EditorState,
  EntryView,
  IBoxLayoutFrameRuntime,
  IBoxLayoutPlugin,
  IBoxLayoutRuntimeAware,
  ILayoutEditor,
  ILayoutEditorEventMap,
  ISubscription,
  LayoutChange,
} from '@chips/sdk';

type GridConfig = {
  columns: number;
  gap: number;
  itemHeight: number;
};

const DEFAULT_CONFIG: GridConfig = {
  columns: 3,
  gap: 16,
  itemHeight: 200,
};

const CONFIG_SCHEMA_VERSION = '1.0.0';

const COLUMNS_RANGE = { min: 1, max: 8 } as const;
const GAP_RANGE = { min: 0, max: 64 } as const;
const ITEM_HEIGHT_RANGE = { min: 80, max: 800 } as const;

let frameRuntime: IBoxLayoutFrameRuntime | null = null;
let activeCleanup: Dispose | null = null;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function readInteger(
  record: Record<string, unknown>,
  key: string,
  fallback: number,
  min: number,
  max: number,
): number {
  const value = record[key];
  if (typeof value !== 'number' || !Number.isFinite(value)) {
    return fallback;
  }
  return Math.min(Math.max(Math.round(value), min), max);
}

function normalizeConfig(input: unknown): GridConfig {
  const record = isRecord(input) ? input : {};
  return {
    columns: readInteger(record, 'columns', DEFAULT_CONFIG.columns, COLUMNS_RANGE.min, COLUMNS_RANGE.max),
    gap: readInteger(record, 'gap', DEFAULT_CONFIG.gap, GAP_RANGE.min, GAP_RANGE.max),
    itemHeight: readInteger(record, 'itemHeight', DEFAULT_CONFIG.itemHeight, ITEM_HEIGHT_RANGE.min, ITEM_HEIGHT_RANGE.max),
  };
}

function isValidField(record: Record<string, unknown>, key: string, min: number, max: number): boolean {
  const value = record[key];
  return value === undefined
    || (typeof value === 'number' && Number.isInteger(value) && value >= min && value <= max);
}

function validateConfig(config: unknown): boolean {
  if (config === undefined || config === null) {
    return true;
  }
  if (!isRecord(config)) {
    return false;
  }
  return (
    isValidField(config, 'columns', COLUMNS_RANGE.min, COLUMNS_RANGE.max)
    && isValidField(config, 'gap', GAP_RANGE.min, GAP_RANGE.max)
    && isValidField(config, 'itemHeight', ITEM_HEIGHT_RANGE.min, ITEM_HEIGHT_RANGE.max)
  );
}

function requireRuntime(): IBoxLayoutFrameRuntime {
  if (frameRuntime === null) {
    throw new Error('帧运行时尚未注入');
  }
  return frameRuntime;
}

function attachRuntime(runtime: IBoxLayoutFrameRuntime): void {
  frameRuntime = runtime;
}

function openEntry(entry: EntryView, fallback: HTMLElement): void {
  void requireRuntime()
    .openDocument(entry.url)
    .catch(() => {
      fallback.textContent = '条目打开失败，请重试';
    });
}

function createTile(entry: EntryView, height: number, observer: IntersectionObserver): HTMLElement {
  const tile = document.createElement('div');
  tile.style.height = `${height}px`;
  tile.style.display = 'flex';
  tile.style.flexDirection = 'column';
  tile.style.border = '1px solid rgba(0, 0, 0, 0.12)';
  tile.style.borderRadius = '8px';
  tile.style.overflow = 'hidden';

  const title = document.createElement('button');
  title.type = 'button';
  title.textContent = entry.displayTitle;
  title.style.textAlign = 'left';
  title.style.padding = '8px 12px';
  title.style.border = '0';
  title.style.background = 'transparent';
  title.style.cursor = 'pointer';
  title.style.font = 'inherit';
  title.addEventListener('click', () => openEntry(entry, tile));

  const slot = document.createElement('div');
  slot.style.flex = '1';
  slot.style.minHeight = '0';
  slot.dataset.entryId = entry.entry_id;

  tile.append(title, slot);
  observer.observe(slot);
  return tile;
}

function createLayout(box: BoxRenderView, container: HTMLElement): Dispose {
  const config = normalizeConfig(box.layout_config);
  const releases = new Map<string, Dispose>();

  const releaseSlot = (entryId: string): void => {
    const release = releases.get(entryId);
    if (release !== undefined) {
      releases.delete(entryId);
      release();
    }
  };

  const mountSlot = (entry: EntryView, slot: HTMLElement): void => {
    if (releases.has(entry.entry_id)) {
      return;
    }
    void requireRuntime()
      .requestEntries({ entryIds: [entry.entry_id] })
      .then(async (tickets) => {
        const ticket = tickets.find((item) => item.entry_id === entry.entry_id);
        if (ticket === undefined || !slot.isConnected) {
          return;
        }
        const release = await requireRuntime().mountEntry(ticket, slot);
        if (!slot.isConnected) {
          release();
          return;
        }
        releases.set(entry.entry_id, release);
      })
      .catch(() => {
        slot.textContent = '条目暂时不可用';
      });
  };

  const observer = new IntersectionObserver((records) => {
    for (const record of records) {
      const slot = record.target as HTMLElement;
      const entryId = slot.dataset.entryId;
      if (entryId === undefined) {
        continue;
      }
      if (record.isIntersecting) {
        const entry = box.entries.find((item) => item.entry_id === entryId);
        if (entry !== undefined) {
          mountSlot(entry, slot);
        }
      } else {
        releaseSlot(entryId);
      }
    }
  });

  const grid = document.createElement('div');
  grid.style.display = 'grid';
  grid.style.gridTemplateColumns = `repeat(${config.columns}, minmax(0, 1fr))`;
  grid.style.gap = `${config.gap}px`;
  grid.style.padding = '16px';
  for (const entry of box.entries) {
    grid.append(createTile(entry, config.itemHeight, observer));
  }

  container.replaceChildren(grid);

  return () => {
    observer.disconnect();
    for (const release of releases.values()) {
      release();
    }
    releases.clear();
    grid.remove();
  };
}

function renderLayout(box: BoxRenderView, container: HTMLElement): Dispose {
  activeCleanup?.();
  const cleanup = createLayout(box, container);
  activeCleanup = cleanup;
  return () => {
    if (activeCleanup === cleanup) {
      activeCleanup = null;
    }
    cleanup();
  };
}

type NumberField = {
  element: HTMLDivElement;
  input: HTMLInputElement;
};

function createNumberField(label: string, value: number, min: number, max: number): NumberField {
  const element = document.createElement('div');
  element.style.display = 'flex';
  element.style.alignItems = 'center';
  element.style.gap = '8px';

  const text = document.createElement('span');
  text.textContent = label;

  const input = document.createElement('input');
  input.type = 'number';
  input.min = String(min);
  input.max = String(max);
  input.value = String(value);

  element.append(text, input);
  return { element, input };
}

class GridEditor implements ILayoutEditor {
  public state: EditorState = 'active';

  private base: GridConfig;
  private revision: number;
  private pending: LayoutChange | null = null;
  private readonly root: HTMLDivElement;
  private readonly columnsField: NumberField;
  private readonly gapField: NumberField;
  private readonly itemHeightField: NumberField;
  private readonly changeHandlers = new Set<(payload: ILayoutEditorEventMap['change']) => void>();
  private readonly dirtyHandlers = new Set<(payload: ILayoutEditorEventMap['dirty']) => void>();

  public constructor(box: BoxRenderView, container: HTMLElement) {
    this.base = normalizeConfig(box.layout_config);
    this.revision = box.revision;

    this.root = document.createElement('div');
    this.root.style.display = 'flex';
    this.root.style.flexDirection = 'column';
    this.root.style.gap = '12px';
    this.root.style.padding = '16px';
    this.root.style.height = '100%';
    this.root.style.boxSizing = 'border-box';

    this.columnsField = createNumberField('列数', this.base.columns, COLUMNS_RANGE.min, COLUMNS_RANGE.max);
    this.gapField = createNumberField('间距', this.base.gap, GAP_RANGE.min, GAP_RANGE.max);
    this.itemHeightField = createNumberField('条目高度', this.base.itemHeight, ITEM_HEIGHT_RANGE.min, ITEM_HEIGHT_RANGE.max);
    this.root.append(this.columnsField.element, this.gapField.element, this.itemHeightField.element);
    container.replaceChildren(this.root);

    this.columnsField.input.addEventListener('input', this.handleInput);
    this.gapField.input.addEventListener('input', this.handleInput);
    this.itemHeightField.input.addEventListener('input', this.handleInput);
  }

  public isDirty(): boolean {
    return this.pending !== null;
  }

  public getChange(): LayoutChange | null {
    return this.pending;
  }

  public markCommitted(revision: number): void {
    if (this.state === 'disposed' || revision <= this.revision) {
      return;
    }
    const committed = this.pending;
    this.revision = revision;
    if (committed !== null && committed.kind === 'replace') {
      this.base = normalizeConfig(committed.config);
    }
    if (this.pending === committed) {
      this.pending = null;
      this.setDirty(false);
    }
  }

  public update(box: BoxRenderView, revision: number): void {
    if (this.state === 'disposed' || revision <= this.revision) {
      return;
    }
    this.revision = revision;
    this.base = normalizeConfig(box.layout_config);
    this.pending = null;
    this.setDirty(false);
    this.renderFromConfig(this.base);
  }

  public subscribe<TEvent extends keyof ILayoutEditorEventMap>(
    event: TEvent,
    handler: (payload: ILayoutEditorEventMap[TEvent]) => void,
  ): ISubscription {
    if (this.state === 'disposed') {
      return { dispose: (): void => undefined };
    }
    if (event === 'change') {
      const typed = handler as (payload: ILayoutEditorEventMap['change']) => void;
      this.changeHandlers.add(typed);
      return {
        dispose: (): void => {
          this.changeHandlers.delete(typed);
        },
      };
    }
    const typed = handler as (payload: ILayoutEditorEventMap['dirty']) => void;
    this.dirtyHandlers.add(typed);
    return {
      dispose: (): void => {
        this.dirtyHandlers.delete(typed);
      },
    };
  }

  public focus(): void {
    if (this.state === 'active') {
      this.columnsField.input.focus();
    }
  }

  public dispose(): void {
    if (this.state === 'disposed') {
      return;
    }
    this.state = 'disposed';
    this.columnsField.input.removeEventListener('input', this.handleInput);
    this.gapField.input.removeEventListener('input', this.handleInput);
    this.itemHeightField.input.removeEventListener('input', this.handleInput);
    this.changeHandlers.clear();
    this.dirtyHandlers.clear();
    this.pending = null;
    this.root.remove();
  }
