function isDocumentClosedPayload(value: unknown): value is IDocumentClosedPayload {
  return isRecord(value) && typeof value.uri === 'string' && isDocumentKind(value.kind);
}

function isDocumentChangedPayload(value: unknown): value is IDocumentChangedPayload {
  return (
    isRecord(value) &&
    typeof value.uri === 'string' &&
    typeof value.documentId === 'string' &&
    isDocumentKind(value.kind) &&
    typeof value.digest === 'string' &&
    typeof value.size === 'number'
  );
}

function isDocumentSavedPayload(value: unknown): value is IDocumentSavedPayload {
  return isRecord(value) && typeof value.uri === 'string' && isDocumentKind(value.kind);
}

function subscribe<TPayload>(
  eventId: string,
  guard: (value: unknown) => value is TPayload,
  handler: (payload: TPayload) => void,
): ISubscription {
  return window.chips.subscribe(eventId, (payload) => {
    if (guard(payload)) {
      handler(payload);
    }
  });
}
