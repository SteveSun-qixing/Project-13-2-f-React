### 6.2 视频卡片（base.video）

```yaml
# base.video 配置 Schema
$schema: "https://json-schema.org/draft/2020-12/schema"
title: "Video Card Config Schema"
type: object
required:
  - card_type
  - video_file
properties:
  card_type:
    const: "base.video"
    description: "视频基础卡片正式类型标识；历史别名 VideoCard 只允许读取归一，不作为新写入值"
  video_file:
    type: string
    description: "视频文件路径（相对于卡片根目录的正式资源路径）"
    examples: ["video.mp4", "travel/demo-video.mp4"]
  cover_image:
    type: string
    description: "封面图片路径（相对于卡片根目录的正式资源路径，可选）"
  subtitles:
    type: array
    description: "外挂字幕资源列表，可为空数组"
    items:
      type: object
      required:
        - id
        - file_path
      properties:
        id:
          type: string
          description: "字幕轨唯一标识"
        label:
          type: string
          description: "字幕轨展示名称，可选"
        language:
          type: string
          description: "BCP 47 语言标识，可选，如 zh-CN / en-US"
        kind:
          type: string
          enum: ["subtitles", "captions"]
          default: "subtitles"
          description: "字幕轨语义类型"
        file_path:
          type: string
          description: "字幕文件路径（相对于卡片根目录的正式资源路径）"
        default:
          type: boolean
          default: false
          description: "是否为默认字幕；同一视频卡片最多一个字幕轨为 true"
  playback:
    type: object
    description: "播放入口参数；播放器应用可按能力选择消费"
    properties:
      autoplay:
        type: boolean
        default: false
      loop:
        type: boolean
        default: false
      muted:
        type: boolean
        default: false
      playback_rate:
        type: number
        minimum: 0.25
        maximum: 4
        default: 1
      start_time:
        type: number
        minimum: 0
        default: 0
  video_title:
    type: string
    description: "视频标题，可选；填写后显示在封面下方"
  publish_time:
    type: string
    description: "发布时间，可选；填写后与创作者一起显示"
  creator:
    type: string
    description: "创作者名称，可选；填写后与发布时间一起显示"
```

示例：

```yaml
# content/f6G7h8I9j0.yaml
card_type: "base.video"
theme: ""
video_file: "travel-video.mp4"
cover_image: "video-cover.jpg"
subtitles:
  - id: "zh-cn"
    label: "简体中文"
    language: "zh-CN"
    kind: "subtitles"
    file_path: "subtitle-zh.vtt"
    default: true
playback:
  autoplay: false
  loop: false
  muted: false
  playback_rate: 1
  start_time: 0
video_title: "东京旅行记录"
publish_time: "2026-04-18"
creator: "薯片工作室"
```
