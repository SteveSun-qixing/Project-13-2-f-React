  public dispose(): void {
    if (this.state === 'disposed') {
      return;
    }
    this.textarea.removeEventListener('input', this.handleInput);
    this.fontSizeInput.removeEventListener('input', this.handleInput);
    this.root.remove();
    this.changeHandlers.clear();
    this.dirtyHandlers.clear();
    this.pending = null;
    this.setDirty(false);
    this.state = 'disposed';
  }

  private readonly handleInput = (): void => {
    if (this.state === 'disposed') {
      return;
    }
    const fontSize = Number(this.fontSizeInput.value);
    const content: Record<string, unknown> = {
      ...this.base.content,
      content_text: this.textarea.value,
      config: {
        ...readConfig(this.base.content),
        fontSize:
          Number.isFinite(fontSize) && fontSize >= MIN_FONT_SIZE && fontSize <= MAX_FONT_SIZE
            ? fontSize
            : DEFAULT_FONT_SIZE,
      },
    };
    const change: BaseCardChange = { kind: 'replace', content };
    this.pending = change;
    for (const handler of this.changeHandlers) {
      handler({ change });
    }
    this.setDirty(true);
  };

  private renderFromView(basecard: BaseCardView): void {
    const config = readConfig(basecard.content);
    this.textarea.value = readText(basecard.content);
    this.fontSizeInput.value = String(config.fontSize);
  }

  private setDirty(isDirty: boolean): void {
    if (this.isDirtyFlag === isDirty) {
      return;
    }
    this.isDirtyFlag = isDirty;
    for (const handler of this.dirtyHandlers) {
      handler({ isDirty });
    }
  }

export function renderEditor(basecard: BaseCardView, container: HTMLElement): IBaseCardEditor {

export async function exportCard(basecard: BaseCardView, format: string): Promise<Blob> {
  if (format !== 'html') {
    throw new ChipsError({
      code: 'CARD_EXPORT_UNSUPPORTED',
      message: `该卡片类型不支持导出为 ${format}`,
      details: { format },
      retryable: false,
    });
  }
  const config = readConfig(basecard.content);
  const body = escapeHtml(readText(basecard.content)).replaceAll('\n', '<br>');
  const html =
    `<article style="font-size: ${config.fontSize}px; line-height: ${config.lineHeight}">` +
    `${body}</article>`;
  return new Blob([html], { type: 'text/html' });
}

### 9.7 对应的卡片内容

`.card/structure.yaml` 声明结构条目，`content/{10 位 Base62 编号}.yaml` 承载条目内容；条目 `type` 与内容 `card_type` 对应：

```yaml
# .card/structure.yaml
structure:
  - id: a1B2c3D4e5
    type: base.richtext
manifest:
  card_count: 1
  resource_count: 0
  resources: []
```

# content/a1B2c3D4e5.yaml
