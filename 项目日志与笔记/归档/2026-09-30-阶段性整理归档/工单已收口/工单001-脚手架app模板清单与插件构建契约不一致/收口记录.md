### 2.1 实测方式（可复现）

`packages/sdk` 构建完成后（`dist/scaffold/index.js` 在位），以纯 Node 动态 `import()` 该 dist 入口，逐角色调用 `createScaffoldPlan()` 打印 `plan.files` 路径序列，并调用 `listScaffoldTemplates()` 打印各模板 `files` 计数：

探针脚本（置于仓库外临时目录，如 `/tmp/t019-work/gen.mjs`；不落库）：

```js
import fs from "node:fs";
const sdkEntry =
  "<工作区绝对路径>/chips-ecosystem/packages/sdk/dist/scaffold/index.js";
const m = await import(sdkEntry);
for (const role of ["basecard", "boxlayout", "app", "service"]) {
  const plan = m.createScaffoldPlan({
    role,
    pluginId: `chips.${role}.t019probe`,
    name: `工单019探针-${role}`,
    targetDir: `/tmp/t019-work/probe-${role}`,
  });
  console.log(
    role,
    plan.files.map((f) => f.path),
  );
}
console.log(
  m.listScaffoldTemplates().map((t) => `${t.role}:${t.files.length}`),
);
```
